// Shared core + submodules. Single schema version and clamp helpers.
export const V1014_SCHEMA_VERSION = '10.14.0';
export const V1014_LIMITS = {
  maxEvidence: 10,
  maxHistory: 16,
  maxDrivers: 8,
  maxRisks: 6,
  hysteresisDeadband: 2.5
} as const;
export function v1014Clamp(n: number, min = 0, max = 100): number {
  if (!Number.isFinite(n)) return min;
  return Math.min(max, Math.max(min, n));
}

// --- 20AG-macro: NATIONAL MACROECONOMY & AGGREGATE BALANCE ---
// ============================================================================
// Synthesizes a read-only national macro view from Living Regions, Industries, Economic Geography,
// Workforce, Public Finance, Households, Systemic Risk and Development Direction — without duplicating
// player cash, Team Treasury, commodity CPI, or fiscal authority (those stay in their owning systems).




export type NationalMacroOutputBand =
  | 'contracting'
  | 'soft'
  | 'moderate'
  | 'expanding'
  | 'surging';

export type NationalMacroCyclePhase =
  | 'slowdown'
  | 'steady'
  | 'expansion'
  | 'overheating'
  | 'adjustment';

export type NationalMacroBalanceBand =
  | 'fragile'
  | 'strained'
  | 'balanced'
  | 'resilient'
  | 'robust';

export interface NationalMacroeconomyEvidence {
  id: string;
  sourceSystem: string;
  sourceId: string;
  factor: string;
  contribution: number;
  regionIds: string[];
  turn: number;
  visibility: 'public';
}

export interface NationalMacroeconomyHistoryEntry {
  turn: number;
  outputBand: NationalMacroOutputBand;
  cyclePhase: NationalMacroCyclePhase;
  summary: string;
}

export interface NationalMacroeconomyPersisted {
  schemaVersion: string;
  revision: number;
  initializedTurn: number | null;
  lastInputHash: string | null;
  lastUpdatedTurn: number;
  outputBand: NationalMacroOutputBand;
  cyclePhase: NationalMacroCyclePhase;
  balanceBand: NationalMacroBalanceBand;
  history: NationalMacroeconomyHistoryEntry[];
}

export interface NationalMacroeconomyState {
  schemaVersion: string;
  turn: number;
  revision: number;
  inputHash: string;
  outputMomentum: number;
  outputBand: NationalMacroOutputBand;
  cyclePhase: NationalMacroCyclePhase;
  balanceBand: NationalMacroBalanceBand;
  diversificationIndex: number;
  externalExposure: number;
  internalDemandStrength: number;
  laborUtilizationProxy: number;
  regionalBalanceIndex: number;
  fiscalCompatibility: number;
  headline: string;
  primaryDriver: string;
  primaryRisk: string | null;
  evidence: NationalMacroeconomyEvidence[];
  drivers: string[];
  risks: string[];
}

export interface NationalMacroeconomyInputs {
  turn: number;
  enabled?: boolean;
  livingRegions?: LivingRegionsState | null;
  industries?: any;
  economicGeography?: EconomicGeographyState | null;
  workforce?: any;
  publicFinance?: NationalPublicFinanceState | null;
  households?: NationalHouseholdEconomyState | null;
  systemicRisk?: any;
  nationalDevelopment?: any;
  highInflationScenarioActive?: boolean;
  activeCrisesCount?: number;
}

export interface NationalMacroeconomyWorldView {
  state: NationalMacroeconomyState;
  persisted: NationalMacroeconomyPersisted;
  inputs: NationalMacroeconomyInputs;
}

function v1014Round(n: number, d = 1): number {
  const p = 10 ** d;
  return Math.round(n * p) / p;
}


function v1014BandFromScore(score: number, bands: Array<[number, NationalMacroOutputBand]>): NationalMacroOutputBand {
  for (const [min, band] of bands) {
    if (score >= min) return band;
  }
  return bands[bands.length - 1][1];
}

const V1014_OUTPUT_BANDS: Array<[number, NationalMacroOutputBand]> = [
  [78, 'surging'],
  [62, 'expanding'],
  [46, 'moderate'],
  [32, 'soft'],
  [0, 'contracting']
];

const V1014_BALANCE_BANDS: Array<[number, NationalMacroBalanceBand]> = [
  [72, 'robust'],
  [58, 'resilient'],
  [44, 'balanced'],
  [30, 'strained'],
  [0, 'fragile']
];

export function createInitialNationalMacroeconomyPersisted(turn = 1): NationalMacroeconomyPersisted {
  return {
    schemaVersion: V1014_SCHEMA_VERSION,
    revision: 0,
    initializedTurn: null,
    lastInputHash: null,
    lastUpdatedTurn: Math.max(1, turn),
    outputBand: 'moderate',
    cyclePhase: 'steady',
    balanceBand: 'balanced',
    history: []
  };
}

export function sanitizeNationalMacroeconomyPersisted(raw: any, turn = 1): NationalMacroeconomyPersisted {
  if (!raw || typeof raw !== 'object') return createInitialNationalMacroeconomyPersisted(turn);
  const hist = Array.isArray(raw.history)
    ? raw.history
        .filter((h: any) => h && typeof h.turn === 'number')
        .slice(-V1014_LIMITS.maxHistory)
        .map((h: any) => ({
          turn: Math.floor(h.turn),
          outputBand: (h.outputBand as NationalMacroOutputBand) || 'moderate',
          cyclePhase: (h.cyclePhase as NationalMacroCyclePhase) || 'steady',
          summary: String(h.summary || '').slice(0, 240)
        }))
    : [];
  return {
    schemaVersion: V1014_SCHEMA_VERSION,
    revision: Math.max(0, Math.floor(Number(raw.revision) || 0)),
    initializedTurn: raw.initializedTurn == null ? null : Math.max(1, Math.floor(Number(raw.initializedTurn) || 1)),
    lastInputHash: typeof raw.lastInputHash === 'string' ? raw.lastInputHash : null,
    lastUpdatedTurn: Math.max(1, Math.floor(Number(raw.lastUpdatedTurn) || turn)),
    outputBand: (raw.outputBand as NationalMacroOutputBand) || 'moderate',
    cyclePhase: (raw.cyclePhase as NationalMacroCyclePhase) || 'steady',
    balanceBand: (raw.balanceBand as NationalMacroBalanceBand) || 'balanced',
    history: hist
  };
}

export function nationalMacroeconomyInputHash(inputs: NationalMacroeconomyInputs): string {
  const lr = inputs.livingRegions;
  const trend = lrNationalTrend(lr);
  const hh = inputs.households?.nationalSummary;
  const pf = inputs.publicFinance?.revenueCapacity;
  const wf = inputs.workforce?.nationalSummary;
  const sr = inputs.systemicRisk?.overallExposure;
  const sig = {
    turn: inputs.turn,
    trend: trend.label,
    growing: trend.growing.length,
    declining: trend.declining.length,
    ind: inputs.industries?.national?.output ? Object.keys(inputs.industries.national.output).length : 0,
    eg: inputs.economicGeography?.nationalSummary?.diversificationScore ?? null,
    hhDemand: hh?.domesticDemandBand ?? null,
    pfCap: pf?.currentCapacity ?? null,
    wfPressure: wf?.overallPressureBand ?? null,
    sr,
    crises: inputs.activeCrisesCount ?? 0,
    infl: Boolean(inputs.highInflationScenarioActive)
  };
  return `v1014:${JSON.stringify(sig)}`;
}

export function buildNationalMacroeconomyInputs(params: {
  turn: number;
  gameState?: any;
  gameSettings?: any;
  livingRegions?: LivingRegionsState | null;
  industries?: any;
  economicGeography?: EconomicGeographyState | null;
  workforce?: any;
  publicFinance?: NationalPublicFinanceState | null;
  households?: NationalHouseholdEconomyState | null;
  systemicRisk?: any;
  nationalDevelopment?: any;
}): NationalMacroeconomyInputs {
  const gs = params.gameState || {};
  const gset = params.gameSettings || {};
  const crisesRaw = gs.crisisChainState?.activeCrisisChains;
  const activeCrisesCount = Array.isArray(crisesRaw)
    ? crisesRaw.filter((c: any) => c && c.status !== 'resolved' && c.status !== 'failed').length
    : 0;
  return {
    turn: Math.max(1, Math.floor(Number(params.turn) || 1)),
    enabled: gset.nationalMacroeconomyEnabled !== false && gset.nationalSystemsEnabled !== false,
    livingRegions: params.livingRegions ?? gs.livingRegionsState ?? gs.livingRegions ?? null,
    industries: params.industries ?? gs.industriesState ?? gs.industries ?? null,
    economicGeography: params.economicGeography ?? gs.economicGeographyState ?? null,
    workforce: params.workforce ?? gs.workforceState ?? null,
    publicFinance: params.publicFinance ?? gs.publicFinanceState ?? null,
    households: params.households ?? gs.householdEconomyState ?? null,
    systemicRisk: params.systemicRisk ?? gs.systemicRiskState ?? null,
    nationalDevelopment: params.nationalDevelopment ?? gs.nationalDevelopmentProfile ?? null,
    highInflationScenarioActive: Boolean(gset.selectedScenarioId === 'high_inflation' || gs.highInflationScenarioActive),
    activeCrisesCount
  };
}

function v1014ApplyHysteresis<T extends string>(raw: T, prev: T | undefined, score: number, upThreshold: number, downThreshold: number): T {
  if (!prev || prev === raw) return raw;
  const db = V1014_LIMITS.hysteresisDeadband;
  if (prev === 'expanding' && raw === 'moderate' && score >= 62 - db) return prev as T;
  if (prev === 'moderate' && raw === 'soft' && score >= 46 - db) return prev as T;
  if (prev === 'surging' && raw === 'expanding' && score >= 78 - db) return prev as T;
  if (score >= upThreshold && raw < prev) return prev as T;
  if (score <= downThreshold && raw > prev) return prev as T;
  return raw;
}

export function computeNationalMacroeconomy(
  inputs: NationalMacroeconomyInputs,
  rawPersisted?: NationalMacroeconomyPersisted | null,
  prevState?: NationalMacroeconomyState | null
): { state: NationalMacroeconomyState; persisted: NationalMacroeconomyPersisted; events: NationalMacroeconomyHistoryEntry[] } {
  const turn = Math.max(1, Math.floor(Number(inputs.turn) || 1));
  const persisted = sanitizeNationalMacroeconomyPersisted(rawPersisted, turn);
  const inputHash = nationalMacroeconomyInputHash(inputs);

  if (persisted.lastInputHash === inputHash && prevState && prevState.inputHash === inputHash) {
    return { state: prevState, persisted, events: [] };
  }

  const evidence: NationalMacroeconomyEvidence[] = [];
  const pushEv = (sourceSystem: string, sourceId: string, factor: string, contribution: number, regionIds: string[] = []) => {
    if (Math.abs(contribution) < 0.05) return;
    evidence.push({
      id: `macro_${sourceSystem}_${sourceId}_${evidence.length}`,
      sourceSystem,
      sourceId,
      factor,
      contribution: v1014Round(contribution),
      regionIds: regionIds.filter(r => V1014_CORE_REGIONS.includes(r)),
      turn,
      visibility: 'public'
    });
  };

  let momentum = 50;
  const drivers: string[] = [];
  const risks: string[] = [];

  const trend = lrNationalTrend(inputs.livingRegions);
  if (trend.label === 'broad national expansion') {
    momentum += 12;
    pushEv('living_regions', 'national_trend', 'Broad regional expansion lifts national output momentum', 12, trend.growing);
    drivers.push('Broad regional expansion');
  } else if (trend.label === 'national slowdown') {
    momentum -= 14;
    pushEv('living_regions', 'national_trend', 'National slowdown weighs on aggregate output', -14, trend.declining);
    risks.push('National slowdown across multiple regions');
  } else if (trend.label === 'modest growth') {
    momentum += 5;
    pushEv('living_regions', 'national_trend', 'Modest national growth', 5, trend.growing);
    drivers.push('Modest national growth');
  } else if (trend.label === 'softening') {
    momentum -= 6;
    pushEv('living_regions', 'national_trend', 'Softening regional momentum', -6, trend.declining);
  }

  const lrRegs = Object.values(inputs.livingRegions?.regions || {});
  if (lrRegs.length) {
    const econAvg =
      lrRegs.reduce((s, r) => s + (r.economy?.score ?? 50), 0) / lrRegs.length;
    const momAvg =
      lrRegs.reduce((s, r) => s + (r.momentum?.value ?? 0), 0) / lrRegs.length;
    const econAdj = (econAvg - 50) * 0.35;
    const momAdj = momAvg * 18;
    momentum += econAdj + momAdj;
    pushEv('living_regions', 'regional_economy', `Average regional economy score ${Math.round(econAvg)}`, econAdj);
    pushEv('living_regions', 'regional_momentum', `Average regional momentum ${v1014Round(momAvg, 2)}`, momAdj);
  }

  let diversificationIndex = 58;
  const indOut: Record<string, number> = inputs.industries?.national?.output || inputs.industries?.output || {};
  const outVals = Object.values(indOut).map(v => Math.max(0, Number(v) || 0));
  const outTotal = outVals.reduce((a, b) => a + b, 0);
  if (outTotal > 0) {
    const shares = outVals.map(v => v / outTotal);
    const hhi = shares.reduce((s, sh) => s + sh * sh, 0);
    diversificationIndex = v1014Clamp(Math.round((1 - hhi) * 140), 18, 95);
    momentum += (diversificationIndex - 50) * 0.12;
    pushEv('industries', 'diversification', `Industrial diversification index ${diversificationIndex}`, (diversificationIndex - 50) * 0.12);
    if (diversificationIndex < 38) risks.push('Concentrated industrial base');
    else if (diversificationIndex >= 72) drivers.push('Diversified industrial base');
  }

  let externalExposure = 42;
  const geoDiv = Number(inputs.economicGeography?.nationalSummary?.diversificationScore ?? inputs.economicGeography?.diversificationScore);
  if (Number.isFinite(geoDiv)) {
    externalExposure = v1014Clamp(Math.round(100 - geoDiv * 0.85), 15, 90);
    pushEv('economic_geography', 'cluster_exposure', 'Geographic concentration affects external exposure', (50 - externalExposure) * 0.08);
  }

  let internalDemandStrength = 52;
  const hhNat = inputs.households?.nationalSummary;
  if (hhNat) {
    const demandMap: Record<string, number> = {
      contracting: 28,
      soft: 38,
      weak: 44,
      moderate: 54,
      strong: 68,
      very_strong: 78,
      surging: 86
    };
    internalDemandStrength = demandMap[String(hhNat.domesticDemandBand)] ?? 52;
    momentum += (internalDemandStrength - 52) * 0.22;
    pushEv('households', 'domestic_demand', `Domestic demand band ${hhNat.domesticDemandBand}`, (internalDemandStrength - 52) * 0.22);
    if (internalDemandStrength >= 68) drivers.push('Strong domestic demand');
    if (internalDemandStrength <= 40) risks.push('Weak domestic demand');
  }

  let laborUtilizationProxy = 55;
  const wfSum = inputs.workforce?.nationalSummary;
  if (wfSum?.overallPressureBand) {
    const wfMap: Record<string, number> = {
      critical_shortage: 82,
      strained: 68,
      tight: 58,
      healthy: 52,
      surplus: 44
    };
    laborUtilizationProxy = wfMap[String(wfSum.overallPressureBand)] ?? 55;
    momentum += (laborUtilizationProxy - 55) * 0.15;
    pushEv('workforce', 'labor_pressure', `Workforce pressure ${wfSum.overallPressureBand}`, (laborUtilizationProxy - 55) * 0.15);
  }

  let fiscalCompatibility = 58;
  const pfCap = inputs.publicFinance?.revenueCapacity;
  if (pfCap) {
    fiscalCompatibility = v1014Clamp(Math.round((Number(pfCap.currentCapacity) + Number(pfCap.resilience)) / 2), 20, 95);
    momentum += (fiscalCompatibility - 58) * 0.1;
    pushEv('public_finance', 'revenue_capacity', 'Public revenue capacity supports macro stability', (fiscalCompatibility - 58) * 0.1);
    if (fiscalCompatibility < 40) risks.push('Thin fiscal capacity for macro shocks');
  }

  const sysExp = Number(inputs.systemicRisk?.overallExposure ?? inputs.systemicRisk?.nationalExposure ?? 0);
  if (sysExp > 0) {
    const adj = -Math.min(18, sysExp * 0.22);
    momentum += adj;
    pushEv('systemic_risk', 'exposure', `Systemic exposure ${Math.round(sysExp)}`, adj);
    if (sysExp >= 55) risks.push('Elevated systemic exposure');
  }

  if ((inputs.activeCrisesCount || 0) > 0) {
    momentum -= Math.min(16, (inputs.activeCrisesCount || 0) * 5);
    pushEv('crises', 'active', `${inputs.activeCrisesCount} active crisis chain(s)`, -Math.min(16, (inputs.activeCrisesCount || 0) * 5));
    risks.push('Active national crises');
  }

  if (inputs.highInflationScenarioActive) {
    momentum -= 8;
    pushEv('scenarios', 'high_inflation', 'High inflation scenario pressure', -8);
    risks.push('Inflation scenario pressure');
  }

  momentum = v1014Clamp(v1014Round(momentum), 8, 92);

  let regionalBalanceIndex = 60;
  if (lrRegs.length >= 4) {
    const scores = lrRegs.map(r => r.economy?.score ?? 50);
    const avg = scores.reduce((a, b) => a + b, 0) / scores.length;
    const dev = Math.sqrt(scores.reduce((s, x) => s + (x - avg) ** 2, 0) / scores.length);
    regionalBalanceIndex = v1014Clamp(Math.round(100 - dev * 1.4), 20, 95);
    if (regionalBalanceIndex < 42) risks.push('Uneven regional development');
  }

  let balanceScore = v1014Clamp(
    Math.round(diversificationIndex * 0.28 + fiscalCompatibility * 0.22 + regionalBalanceIndex * 0.2 + (100 - externalExposure) * 0.15 + (100 - sysExp) * 0.15),
    10,
    95
  );

  const rawOutputBand = v1014BandFromScore(momentum, V1014_OUTPUT_BANDS);
  const outputBand = v1014ApplyHysteresis(rawOutputBand, persisted.outputBand, momentum, 62, 46) as NationalMacroOutputBand;

  let cyclePhase: NationalMacroCyclePhase = 'steady';
  if (momentum >= 74 && internalDemandStrength >= 65 && laborUtilizationProxy >= 62) cyclePhase = 'overheating';
  else if (momentum >= 62) cyclePhase = 'expansion';
  else if (momentum <= 34) cyclePhase = 'slowdown';
  else if (momentum <= 42 && sysExp >= 45) cyclePhase = 'adjustment';
  if (persisted.cyclePhase === 'expansion' && cyclePhase === 'steady' && momentum >= 58 - V1014_LIMITS.hysteresisDeadband) {
    cyclePhase = 'expansion';
  }

  const rawBalanceBand = v1014BandFromScore(balanceScore, V1014_BALANCE_BANDS);
  const balanceBand = v1014ApplyHysteresis(rawBalanceBand, persisted.balanceBand, balanceScore, 58, 44) as NationalMacroBalanceBand;

  const headline =
    outputBand === 'surging'
      ? 'National economy surging with broad momentum'
      : outputBand === 'expanding'
      ? 'National economy expanding'
      : outputBand === 'contracting'
      ? 'National economy contracting'
      : outputBand === 'soft'
      ? 'National economy softening'
      : 'National economy growing at a moderate pace';

  const primaryDriver = drivers[0] || (trend.label !== 'steady' ? `Regional trend: ${trend.label}` : 'Balanced regional activity');
  const primaryRisk = risks[0] || null;

  const state: NationalMacroeconomyState = {
    schemaVersion: V1014_SCHEMA_VERSION,
    turn,
    revision: persisted.revision + (persisted.lastInputHash === inputHash ? 0 : 1),
    inputHash,
    outputMomentum: momentum,
    outputBand,
    cyclePhase,
    balanceBand,
    diversificationIndex,
    externalExposure,
    internalDemandStrength,
    laborUtilizationProxy,
    regionalBalanceIndex,
    fiscalCompatibility,
    headline,
    primaryDriver,
    primaryRisk,
    evidence: evidence.slice(-V1014_LIMITS.maxEvidence),
    drivers: drivers.slice(0, V1014_LIMITS.maxDrivers),
    risks: risks.slice(0, V1014_LIMITS.maxRisks)
  };

  const events: NationalMacroeconomyHistoryEntry[] = [];
  const bandChanged = outputBand !== persisted.outputBand || cyclePhase !== persisted.cyclePhase;
  const nextHistory = [...persisted.history];
  if (bandChanged || persisted.initializedTurn === null) {
    const entry: NationalMacroeconomyHistoryEntry = {
      turn,
      outputBand,
      cyclePhase,
      summary: `${headline} (${cyclePhase} phase)`
    };
    nextHistory.push(entry);
    events.push(entry);
  }

  const nextPersisted: NationalMacroeconomyPersisted = {
    ...persisted,
    revision: state.revision,
    initializedTurn: persisted.initializedTurn ?? turn,
    lastInputHash: inputHash,
    lastUpdatedTurn: turn,
    outputBand,
    cyclePhase,
    balanceBand,
    history: nextHistory.slice(-V1014_LIMITS.maxHistory)
  };

  return { state, persisted: nextPersisted, events };
}

export function validateNationalMacroeconomyState(state: NationalMacroeconomyState | null | undefined): { valid: boolean; issues: string[] } {
  const issues: string[] = [];
  if (!state) return { valid: false, issues: ['NationalMacroeconomyState is null or undefined'] };
  if (state.schemaVersion !== V1014_SCHEMA_VERSION) issues.push(`Unexpected schema ${state.schemaVersion}`);
  for (const k of ['outputMomentum', 'diversificationIndex', 'externalExposure', 'internalDemandStrength'] as const) {
    const v = state[k];
    if (!Number.isFinite(v)) issues.push(`Non-finite ${k}`);
  }
  return { valid: issues.length === 0, issues };
}

/** GI3 / contextual ranking: small bounded macro fit (≤ ±2.0). */
export function nationalMacroeconomyActionAlignment(
  macro: NationalMacroeconomyState | null | undefined,
  action: { category?: string; regionId?: string; title?: string; description?: string }
): { scoreDelta: number; reason: string | null } {
  if (!macro) return { scoreDelta: 0, reason: null };
  const blob = `${action.category || ''} ${action.title || ''} ${action.description || ''}`.toLowerCase();
  let delta = 0;
  if (macro.outputBand === 'contracting' || macro.cyclePhase === 'slowdown') {
    if (/recovery|resilien|stabil|relief|diversif/.test(blob)) delta += 1.5;
    if (/speculative|leverag|overextend/.test(blob)) delta -= 1.2;
  }
  if (macro.cyclePhase === 'overheating' && /housing|overheat|speculative/.test(blob)) delta -= 1.4;
  if (macro.balanceBand === 'fragile' && /resilien|redundan|backup|diversif/.test(blob)) delta += 1.6;
  if (macro.internalDemandStrength >= 68 && /manufactur|domestic|workforce|skills/.test(blob)) delta += 0.8;
  delta = v1014Clamp(delta, -2, 2);
  return { scoreDelta: v1014Round(delta), reason: delta !== 0 ? macro.headline : null };
}

export type NationalMacroeconomyQueryTopic = 'overview' | 'momentum' | 'cycle' | 'balance' | 'diversification' | 'risk';

export function detectNationalMacroeconomyQuery(raw: string): NationalMacroeconomyQueryTopic | null {
  const q = ` ${String(raw || '').toLowerCase().replace(/[’']/g, "'").replace(/[?!.]/g, ' ').replace(/\s+/g, ' ').trim()} `;
  if (!/\b(national economy|macroeconom|macro economy|aggregate economy|overall economy|gdp|economic cycle|business cycle)\b/.test(q)) return null;
  if (/\b(diversif|concentrat|depend)\b/.test(q)) return 'diversification';
  if (/\b(risk|fragile|vulnerab|exposure)\b/.test(q)) return 'risk';
  if (/\b(cycle|phase|overheat|slowdown|expansion)\b/.test(q)) return 'cycle';
  if (/\b(momentum|growth|contract|surge|soften)\b/.test(q)) return 'momentum';
  if (/\b(balance|resilien|robust|strained)\b/.test(q)) return 'balance';
  return 'overview';
}

export function answerNationalMacroeconomyQuestion(
  state: NationalMacroeconomyState | null | undefined,
  topic: NationalMacroeconomyQueryTopic
): { fact: string; calculated: string; inference: string; projection: string } {
  if (!state) {
    return {
      fact: 'National macroeconomy layer is off or not yet computed.',
      calculated: 'Enable National Systems and V10.14 National Macroeconomy in settings.',
      inference: 'No macro synthesis is available this turn.',
      projection: 'Turn the layer on to see aggregate momentum and balance.'
    };
  }
  const fact = `${state.headline} Output momentum ${state.outputMomentum}/100 (${state.outputBand.replace(/_/g, ' ')}).`;
  const calculated = `Cycle phase: ${state.cyclePhase.replace(/_/g, ' ')} · Balance: ${state.balanceBand.replace(/_/g, ' ')} · Diversification ${state.diversificationIndex}/100 · Domestic demand strength ${state.internalDemandStrength}/100.`;
  const inference = state.primaryRisk
    ? `Primary macro risk: ${state.primaryRisk}. Main driver: ${state.primaryDriver}.`
    : `Main driver: ${state.primaryDriver}. No dominant macro risk flagged this turn.`;
  const projection =
    state.cyclePhase === 'overheating'
      ? 'Without cooling investment or capacity relief, overheating could tighten labor and fiscal headroom.'
      : state.cyclePhase === 'slowdown'
      ? 'Recovery-oriented infrastructure and diversification would improve macro momentum over several turns.'
      : 'Current macro balance suggests steady pacing — watch regional divergence and systemic exposure.';
  if (topic === 'diversification') {
    return { fact, calculated: `Diversification index ${state.diversificationIndex}/100 · External exposure ${state.externalExposure}/100.`, inference, projection };
  }
  if (topic === 'risk') {
    return { fact, calculated: `Systemic macro risks: ${state.risks.length ? state.risks.join('; ') : 'none flagged'}.`, inference, projection };
  }
  if (topic === 'cycle') {
    return { fact, calculated: `Business-cycle phase: ${state.cyclePhase.replace(/_/g, ' ')} (output band ${state.outputBand}).`, inference, projection };
  }
  if (topic === 'momentum') {
    return { fact, calculated: `Output momentum ${state.outputMomentum}/100 with ${state.drivers.slice(0, 2).join(' · ') || 'mixed drivers'}.`, inference, projection };
  }
  if (topic === 'balance') {
    return { fact, calculated: `Macro balance band ${state.balanceBand} (regional balance ${state.regionalBalanceIndex}/100, fiscal compatibility ${state.fiscalCompatibility}/100).`, inference, projection };
  }
  return { fact, calculated, inference, projection };
}

export function v1014NormalizeCommodityKey(key: string): string {
  return String(key || '')
    .trim()
    .toLowerCase()
    .replace(/\s+/g, '_');
}

/** Part 17 boundary — commodities that must never enter household CPI averaging. */
export function isNonCpiCommodity(resourceKey: string): boolean {
  const norm = v1014NormalizeCommodityKey(resourceKey);
  return V1013_NON_CPI_COMMODITIES.some((c) => v1014NormalizeCommodityKey(c) === norm);
}


function scoreToHeadlineBand(score: number): HeadlineInflationBand {
  if (score >= 82) return 'critical';
  if (score >= 68) return 'high';
  if (score >= 52) return 'elevated';
  if (score >= 36) return 'stable';
  return 'subdued';
}

export function buildCpiBasketMetrics(
  current: Record<string, number> | null | undefined,
  base: Record<string, number> | null | undefined
): {
  averageRatio: number;
  sampleCount: number;
  excludedSpikeCount: number;
  nonCpiBoundaryIntact: boolean;
} {
  const cur = current || {};
  const bas = base || {};
  let sum = 0;
  let count = 0;
  let excludedSpikeCount = 0;
  let boundaryViolations = 0;

  const keys = new Set([...Object.keys(cur), ...Object.keys(bas)]);
  for (const key of keys) {
    if (isNonCpiCommodity(key)) {
      const b = bas[key];
      const c = cur[key];
      if (typeof b === 'number' && typeof c === 'number' && b > 0 && c / b > 1.35) {
        excludedSpikeCount += 1;
      }
      continue;
    }
    const basePrice = bas[key];
    const currPrice = cur[key];
    if (typeof basePrice !== 'number' || typeof currPrice !== 'number' || !Number.isFinite(basePrice) || basePrice <= 0) {
      continue;
    }
    if (isNonCpiCommodity(key)) {
      boundaryViolations += 1;
      continue;
    }
    sum += currPrice / basePrice;
    count += 1;
  }

  const averageRatio = count > 0 ? sum / count : 1;
  return {
    averageRatio,
    sampleCount: count,
    excludedSpikeCount,
    nonCpiBoundaryIntact: boundaryViolations === 0
  };
}

export function nationalInflationInputHash(inputs: NationalInflationInputs): string {
  const crisisDigest = (inputs.crises || [])
    .map((c) => `${c.id || ''}:${c.category || ''}:${c.severity || ''}`)
    .sort()
    .join(',');
  const priceDigest = Object.entries(inputs.resourcePrices || {})
    .filter(([k]) => !isNonCpiCommodity(k))
    .slice(0, 24)
    .map(([k, v]) => `${k}:${Math.round(Number(v) || 0)}`)
    .sort()
    .join(',');
  const pf = inputs.publicFinance;
  const pfDigest = pf ? `${pf.fiscalPosition?.pressure || ''}:${pf.fiscalPosition?.fiscalHeadroom ?? ''}` : '';
  const raw = [
    `t${inputs.turn}`,
    `en${inputs.enabled}`,
    `sc${inputs.scenarioId || ''}`,
    `im${inputs.inflationMultiplier ?? 1}`,
    `hi${inputs.highInflationScenarioFlag ? 1 : 0}`,
    `mt${inputs.marketTrend || ''}`,
    `tc${inputs.turnCounter ?? 0}`,
    `e${inputs.energySupplyHealth ?? ''}`,
    `a${inputs.agricultureSupplyHealth ?? ''}`,
    `tr${inputs.transportNetworkHealth ?? ''}`,
    `sr${inputs.systemicRiskPressure ?? ''}`,
    `pf${pfDigest}`,
    `pr${priceDigest}`,
    `cr${crisisDigest}`
  ].join('||');
  let h = 2166136261;
  for (let i = 0; i < raw.length; i++) {
    h ^= raw.charCodeAt(i);
    h = Math.imul(h, 16777619);
  }
  return `hi1014_${(h >>> 0).toString(36)}`;
}

export function buildNationalInflationInputs(params: any): NationalInflationInputs {
  const gs = params?.gameState || params || {};

  const scenarioId = String(gset.selectedScenarioId || gs.selectedScenarioId || '').toLowerCase();
  const inflationMultiplier = Number(gset.inflationMultiplier ?? gs.inflationMultiplier ?? 1);
  const highInflationScenarioFlag =
    scenarioId.includes('inflation') ||
    Boolean(gs.scenarioModifiers?.highInflation) ||

  const basePrices: Record<string, number> = {};

  for (const r of resources) {

  }

  return {

    scenarioId,
    inflationMultiplier,
    highInflationScenarioFlag,
    resourcePrices: gs.resourcePrices || params?.resourcePrices || null,
    baseResourcePrices: params?.baseResourcePrices || basePrices,
    marketTrend: gs.marketTrend || gs.resourceMarketTrend || null,

    agricultureSupplyHealth: params?.agricultureSupplyHealth,

  };
}

export function computeNationalInflation(
  inputs: NationalInflationInputs,
  previous: NationalInflationState | null
): NationalInflationState | null {
  if (!inputs.enabled) return null;

  const turn = inputs.turn;
  const evidence: InflationEvidence[] = [];
  const pushEv = (sourceSystem: string, factor: string, contribution: number) => {
    if (evidence.length >= V1014_LIMITS.maxEvidence) return;
    evidence.push({
      id: `ev_${sourceSystem}_${factor}_${turn}_${evidence.length}`,
      sourceSystem,
      factor,
      contribution,
      turn
    });
  };

  let headline = 42;
  const basket = buildCpiBasketMetrics(inputs.resourcePrices, inputs.baseResourcePrices);
  if (basket.sampleCount > 0) {
    const ratioPressure = (basket.averageRatio - 1) * 120;
    headline += ratioPressure;
    pushEv('resource_market', 'cpi_basket_ratio', Math.round(ratioPressure));
  }
  if (basket.excludedSpikeCount > 0) {
    pushEv('resource_market', 'non_cpi_spike_excluded', 0);
  }

  const trend = String(inputs.marketTrend || 'stable').toLowerCase();
  if (trend === 'rising') {
    headline += 6;
    pushEv('resource_market', 'market_trend_rising', 6);
  } else if (trend === 'volatile') {
    headline += 8;
    pushEv('resource_market', 'market_trend_volatile', 8);
  } else if (trend === 'falling') {
    headline -= 4;
    pushEv('resource_market', 'market_trend_falling', -4);
  }

  const tc = Math.max(0, Number(inputs.turnCounter ?? turn) || 0);
  const drift = Math.min(12, tc * 0.35);
  headline += drift;
  pushEv('item_pricing', 'turn_drift_proxy', Math.round(drift));

  const energy = inputs.energySupplyHealth ?? 70;
  const ag = inputs.agricultureSupplyHealth ?? 70;
  const transport = inputs.transportNetworkHealth ?? 70;
  if (energy < 55) {
    const d = Math.round((55 - energy) * 0.45);
    headline += d;
    pushEv('supply', 'energy_strain', d);
  }
  if (ag < 55) {
    const d = Math.round((55 - ag) * 0.42);
    headline += d;
    pushEv('supply', 'agriculture_strain', d);
  }
  if (transport < 55) {
    const d = Math.round((55 - transport) * 0.38);
    headline += d;
    pushEv('supply', 'transport_strain', d);
  }

  const pfPressure = inputs.publicFinance?.fiscalPosition?.pressure;
  if (pfPressure === 'strained' || pfPressure === 'critical') {
    headline += pfPressure === 'critical' ? 10 : 6;
    pushEv('public_finance', `fiscal_${pfPressure}`, pfPressure === 'critical' ? 10 : 6);
  } else if (pfPressure === 'tight') {
    headline += 3;
    pushEv('public_finance', 'fiscal_tight', 3);
  }

  const sr = inputs.systemicRiskPressure ?? 0;
  if (sr >= 45) {
    const d = Math.round((sr - 40) * 0.2);
    headline += d;
    pushEv('systemic_risk', 'risk_pressure', d);
  }

  const scenarioInfl =
    Boolean(inputs.highInflationScenarioFlag) || Number(inputs.inflationMultiplier || 1) > 1.15;
  if (scenarioInfl) {
    headline += 14;
    pushEv('scenario', 'high_inflation_preset', 14);
  }

  const commoditySpikeActive = basket.excludedSpikeCount > 0;

  headline = v1014Clamp(headline);

  let headlineBand = scoreToHeadlineBand(headline);
  if (previous && previous.headlineBand) {
    const prevRank = HEADLINE_INFLATION_RANK[previous.headlineBand];
    const nextRank = HEADLINE_INFLATION_RANK[headlineBand];
    const delta = headline - (previous.headlineIndex ?? headline);
    const bandLadder: HeadlineInflationBand[] = ['subdued', 'stable', 'elevated', 'high', 'critical'];
    if (Math.abs(delta) < V1014_LIMITS.headlineHysteresisDeadband) {
      if (nextRank > prevRank + 1) headlineBand = bandLadder[prevRank + 1];
      if (nextRank < prevRank - 1) headlineBand = bandLadder[prevRank - 1];
    }
  }

  const highInflationScenarioActive =
    scenarioInfl || headlineBand === 'high' || headlineBand === 'critical' || headline >= 68;

  const categoryBase = {
    essentials: headline * 0.92,
    services: headline * 0.88,
    housing: headline * 0.95,
    energy: headline * 0.9 + Math.max(0, 55 - energy) * 0.35,
    transport: headline * 0.86 + Math.max(0, 55 - transport) * 0.3,
    food: headline * 0.84 + Math.max(0, 55 - ag) * 0.32
  };

  if (highInflationScenarioActive) {
    categoryBase.essentials += 8;
    categoryBase.services += 10;
  }

  const categoryPressures: InflationCategoryPressure[] = (Object.keys(categoryBase) as InflationPriceCategory[]).map(
    (category) => {
      const pressureScore = v1014Clamp(categoryBase[category]);
      return { category, pressureScore, band: scoreToCostPressureBand(pressureScore) };
    }
  );

  const itemPriceDriftMultiplier = Math.max(0.85, Math.min(1.45, 1 + (headline - 42) / 200 + tc * 0.003));
  const resourceMarketPressureMultiplier = Math.max(
    0.9,
    Math.min(1.25, 1 + (basket.sampleCount > 0 ? (basket.averageRatio - 1) * 0.35 : 0))
  );

  const headlineYoYProxyPct = Math.max(-2, Math.min(18, Math.round((headline - 42) * 0.22 + (scenarioInfl ? 4 : 0))));

  const inputHash = nationalInflationInputHash(inputs);

  return {
    schemaVersion: V1014_SCHEMA_VERSION,
    revision: (previous?.revision ?? 0) + 1,
    lastUpdatedTurn: turn,
    inputHash,
    headlineIndex: headline,
    headlineBand,
    headlineYoYProxyPct,
    highInflationScenarioActive,
    commoditySpikeActive,
    nonCpiBoundaryIntact: basket.nonCpiBoundaryIntact,
    cpiBasketSampleCount: basket.sampleCount,
    categoryPressures,
    itemPriceDriftMultiplier,
    resourceMarketPressureMultiplier,
    evidence
  };
}

export function validateNationalInflationState(state: NationalInflationState | null | undefined): {
  valid: boolean;
  issues: string[];
} {
  if (!state) return { valid: true, issues: [] };
  const issues: string[] = [];
  if (state.schemaVersion !== V1014_SCHEMA_VERSION) issues.push('schemaVersion mismatch');
  if (!Number.isFinite(state.headlineIndex) || state.headlineIndex < 0 || state.headlineIndex > 100) {
    issues.push('headlineIndex out of bounds');
  }
  if (!HEADLINE_INFLATION_LABEL[state.headlineBand]) issues.push('invalid headlineBand');
  if (state.categoryPressures.length !== 6) issues.push('categoryPressures incomplete');
  if (state.commoditySpikeActive && !state.nonCpiBoundaryIntact) {
    /* spike allowed when boundary intact */
  }
  return { valid: issues.length === 0, issues };
}

  | 'distressed'
  | 'cautious'
  | 'neutral'
  | 'supportive'
  | 'expansionary';

  | 'contracting'
  | 'weak'
  | 'stable'
  | 'improving'
  | 'strong';

  | 'fragile'
  | 'low'
  | 'balanced'
  | 'confident'
  | 'very_confident';

const V1014_CLIMATE_RANK: Record<CorporateClimateBand, number> = {
  distressed: 0,
  cautious: 1,
  neutral: 2,
  supportive: 3,
  expansionary: 4
};

const V1014_OUTLOOK_RANK: Record<FirmProfitOutlookBand, number> = {
  contracting: 0,
  weak: 1,
  stable: 2,
  improving: 3,
  strong: 4
};

const V1014_CONF_RANK: Record<CorporateConfidenceBand, number> = {
  fragile: 0,
  low: 1,
  balanced: 2,
  confident: 3,
  very_confident: 4
};

const V1014_CLIMATE_ORDER: CorporateClimateBand[] = ['distressed', 'cautious', 'neutral', 'supportive', 'expansionary'];
const V1014_OUTLOOK_ORDER: FirmProfitOutlookBand[] = ['contracting', 'weak', 'stable', 'improving', 'strong'];
const V1014_CONF_ORDER: CorporateConfidenceBand[] = ['fragile', 'low', 'balanced', 'confident', 'very_confident'];

const v1014CorporateRound = (x: number) => Math.round((Number.isFinite(x) ? x : 0) * 10) / 10;

export function createEmptyCorporateSectorPersisted(): CorporateSectorPersisted {
  return {
    schemaVersion: '10.14',
    revision: 0,
    initializedTurn: null,
    lastUpdatedTurn: 0,
    climateBands: {},
    outlookBands: {},
    confidenceBands: {},
    history: [],
    inputHash: ''
  };
}

export function sanitizeCorporateSectorPersisted(raw: unknown, turn = 1): CorporateSectorPersisted {
  const base = createEmptyCorporateSectorPersisted();
  if (!raw || typeof raw !== 'object') return base;
  const src = raw as Partial<CorporateSectorPersisted>;
  const pickBand = <T extends string>(v: unknown, allowed: readonly T[], fallback: T): T =>
    typeof v === 'string' && (allowed as readonly string[]).includes(v) ? (v as T) : fallback;
  const climateBands: Record<string, CorporateClimateBand> = {};
  const outlookBands: Record<string, FirmProfitOutlookBand> = {};
  const confidenceBands: Record<string, CorporateConfidenceBand> = {};
  if (src.climateBands && typeof src.climateBands === 'object') {
    for (const [k, v] of Object.entries(src.climateBands)) {
      climateBands[k] = pickBand(v, V1014_CLIMATE_ORDER, 'neutral');
    }
  }
  if (src.outlookBands && typeof src.outlookBands === 'object') {
    for (const [k, v] of Object.entries(src.outlookBands)) {
      outlookBands[k] = pickBand(v, V1014_OUTLOOK_ORDER, 'stable');
    }
  }
  if (src.confidenceBands && typeof src.confidenceBands === 'object') {
    for (const [k, v] of Object.entries(src.confidenceBands)) {
      confidenceBands[k] = pickBand(v, V1014_CONF_ORDER, 'balanced');
    }
  }
  const history = Array.isArray(src.history)
    ? src.history.slice(0, V1014_LIMITS.history).map((h, i) => ({
        id: typeof (h as any)?.id === 'string' ? (h as any).id : `ce_hist_${turn}_${i}`,
        turn: Math.max(0, Math.floor(Number((h as any)?.turn) || turn)),
        kind: (['climate_shifted', 'outlook_weakened', 'outlook_strengthened', 'confidence_shifted'] as const).includes((h as any)?.kind)
          ? (h as any).kind
          : 'climate_shifted',
        fromBand: String((h as any)?.fromBand || ''),
        toBand: String((h as any)?.toBand || ''),
        summary: String((h as any)?.summary || '').slice(0, 240)
      }))
    : [];
  return {
    schemaVersion: '10.14',
    revision: Math.max(0, Math.floor(Number(src.revision) || 0)),
    initializedTurn: src.initializedTurn === null || src.initializedTurn === undefined ? null : Math.max(0, Math.floor(Number(src.initializedTurn))),
    lastUpdatedTurn: Math.max(0, Math.floor(Number(src.lastUpdatedTurn) || turn)),
    climateBands,
    outlookBands,
    confidenceBands,
    history,
    inputHash: typeof src.inputHash === 'string' ? src.inputHash : ''
  };
}

export function corporateSectorInputHash(inputs: CorporateSectorInputs): string {
  const ind = inputs.industries;
  const sc = inputs.strategicCapital;
  const hh = inputs.householdEconomy;
  const raw = JSON.stringify([
    inputs.turn,
    inputs.enabled,
    ind ? [ind.revision, ind.lastUpdatedTurn, ind.national?.primaryIndustrialRisk, ind.national?.constrainedIndustries?.join(',')] : null,
    sc ? [sc.revision, sc.investmentClimate?.overallClimate, sc.investmentClimate?.debtEnvironmentPressure] : null,
    hh ? [hh.revision, hh.domesticDemand?.level, hh.national?.prosperity] : null
  ]);
  let h = 2166136261;
  for (let k = 0; k < raw.length; k++) {
    h ^= raw.charCodeAt(k);
    h = Math.imul(h, 16777619);
  }
  return (h >>> 0).toString(36);
}

function v1014CorporateBandFromScore(score: number, bands: Array<{ min: number; band: CorporateClimateBand }>): CorporateClimateBand {
  for (let i = bands.length - 1; i >= 0; i--) {
    if (score >= bands[i].min) return bands[i].band;
  }
  return bands[0]?.band || 'neutral';
}

function v1014OutlookFromScore(score: number): FirmProfitOutlookBand {
  if (score < 25) return 'contracting';
  if (score < 40) return 'weak';
  if (score < 58) return 'stable';
  if (score < 72) return 'improving';
  return 'strong';
}

function v1014ConfidenceFromScore(score: number): CorporateConfidenceBand {
  if (score < 22) return 'fragile';
  if (score < 38) return 'low';
  if (score < 55) return 'balanced';
  if (score < 72) return 'confident';
  return 'very_confident';
}

function v1014ApplyHysteresisClimate(prev: CorporateClimateBand | undefined, raw: CorporateClimateBand, score: number): CorporateClimateBand {
  if (!prev || prev === raw) return raw;
  const delta = V1014_CLIMATE_RANK[raw] - V1014_CLIMATE_RANK[prev];
  if (Math.abs(delta) <= 1) return raw;
  const deadband = V1014_LIMITS.hysteresisDeadband;
  const thresholds: Record<CorporateClimateBand, number> = {
    distressed: 18,
    cautious: 32,
    neutral: 48,
    supportive: 62,
    expansionary: 78
  };
  if (delta > 0 && score < thresholds[raw] + deadband) return prev;
  if (delta < 0 && score > thresholds[prev] - deadband) return prev;
  return raw;
}

function scoreIndustryHealth(ind: IndustriesSupplyChainsState | null): { score: number; constraint: string | null; evidence: CorporateSectorEvidence[] } {
  if (!ind) return { score: 50, constraint: null, evidence: [{ label: 'Industry layer inactive', value: 50, source: 'v1014_default' }] };
  const nat = ind.national;
  const constrained = nat.constrainedIndustries?.length || 0;
  const strong = nat.strongestIndustries?.length || 0;
  const shortages = nat.shortages?.length || 0;
  let score = 55 + strong * 6 - constrained * 7 - shortages * 5;
  score = v1014Clamp(score, 5, 95);
  const constraint = nat.primaryIndustrialRisk || (shortages > 0 ? `Supply shortages (${shortages})` : null);
  return {
    score,
    constraint,
    evidence: [
      { label: 'Strong industries', value: strong, source: 'v101_industries' },
      { label: 'Constrained industries', value: constrained, source: 'v101_industries' },
      { label: 'Active shortages', value: shortages, source: 'v101_industries' }
    ]
  };
}

function scoreCapitalAlignment(sc: StrategicCapitalState | null): { score: number; evidence: CorporateSectorEvidence[] } {
  if (!sc?.investmentClimate) {
    return { score: 50, evidence: [{ label: 'Capital climate unavailable', value: 50, source: 'v1014_default' }] };
  }
  const ic = sc.investmentClimate;
  const climateBoost: Record<string, number> = {
    crisis_defensive: -18,
    constrained: -10,
    selective: 0,
    opportunistic: 8,
    expansionary: 14
  };
  const base = 52 + (climateBoost[ic.overallClimate] || 0);
  const pressurePenalty = v1014Clamp((ic.debtEnvironmentPressure || 0) / 4, 0, 18);
  const score = v1014Clamp(base - pressurePenalty + (ic.innovationOpportunity || 0) / 15, 8, 92);
  return {
    score,
    evidence: [
      { label: 'Investment climate', value: climateBoost[ic.overallClimate] || 0, source: 'v109_climate' },
      { label: 'Debt environment pressure', value: ic.debtEnvironmentPressure || 0, source: 'v109_climate' }
    ]
  };
}

function scoreDemandSupport(hh: NationalHouseholdEconomyState | null): { score: number; evidence: CorporateSectorEvidence[] } {
  if (!hh?.domesticDemand) {
    return { score: 50, evidence: [{ label: 'Domestic demand unavailable', value: 50, source: 'v1014_default' }] };
  }
  const rank: Record<string, number> = {
    contracting: 20,
    soft: 32,
    weak: 40,
    moderate: 52,
    strong: 68,
    very_strong: 78,
    surging: 88
  };
  const score = rank[hh.domesticDemand.level] ?? 50;
  return {
    score,
    evidence: [{ label: 'Domestic demand band', value: score, source: 'v1013_households' }]
  };
}

function buildRegionalOutlooks(
  ind: IndustriesSupplyChainsState | null,
  national: NationalCorporateSectorSummary,
  persisted: CorporateSectorPersisted,
  turn: number
): Record<string, RegionalCorporateOutlook> {
  const out: Record<string, RegionalCorporateOutlook> = {};
  if (!ind?.regions) return out;
  for (const [regionId, reg] of Object.entries(ind.regions).slice(0, V1014_LIMITS.maxRegions)) {
    const inds = reg.industries || {};
    const conditions = Object.values(inds).map((s) => s?.condition).filter(Boolean) as IndustryCondition[];
    const constrained = conditions.filter((c) => c === 'constrained' || c === 'disrupted').length;
    const strong = conditions.filter((c) => c === 'strong' || c === 'booming').length;
    let score = national.compositeScore * 0.55 + strong * 8 - constrained * 9;
    score = v1014Clamp(score, 5, 95);
    const rawClimate = v1014CorporateBandFromScore(score, [
      { min: 78, band: 'expansionary' },
      { min: 62, band: 'supportive' },
      { min: 45, band: 'neutral' },
      { min: 28, band: 'cautious' },
      { min: 0, band: 'distressed' }
    ]);
    const climate = v1014ApplyHysteresisClimate(persisted.climateBands[regionId], rawClimate, score);
    const profitOutlook = v1014OutlookFromScore(score * 0.85 + national.demandSupportScore * 0.15);
    const confidence = v1014ConfidenceFromScore((score + national.capitalAlignmentScore) / 2);
    out[regionId] = {
      regionId,
      climate,
      profitOutlook,
      confidence,
      primaryConstraint: reg.primaryIndustrialConstraint,
      score: v1014CorporateRound(score),
      evidence: [
        { label: 'Regional strong industries', value: strong, source: 'v101_regional' },
        { label: 'Regional constrained', value: constrained, source: 'v101_regional' }
      ]
    };
  }
  return out;
}

export function computeCorporateSectorEconomy(
  inputs: CorporateSectorInputs,
  persistedRaw: CorporateSectorPersisted | null | undefined,
  _prevState: CorporateSectorState | null
): { state: CorporateSectorState; persisted: CorporateSectorPersisted; derived: CorporateSectorHistoryEntry[] } {
  const t0 = typeof performance !== 'undefined' ? performance.now() : Date.now();
  const persisted = sanitizeCorporateSectorPersisted(persistedRaw, inputs.turn);
  const hash = corporateSectorInputHash(inputs);

  if (!inputs.enabled) {
    const empty: CorporateSectorState = {
      schemaVersion: '10.14',
      revision: persisted.revision,
      enabled: false,
      turn: inputs.turn,
      national: {
        climate: 'neutral',
        profitOutlook: 'stable',
        confidence: 'balanced',
        industryHealthScore: 50,
        capitalAlignmentScore: 50,
        demandSupportScore: 50,
        compositeScore: 50,
        dominantConstraint: null,
        narrative: 'Corporate sector interpretation disabled.',
        evidence: []
      },
      regions: {},
      history: persisted.history,
      inputHash: hash,
      computeMs: 0
    };
    return { state: empty, persisted: { ...persisted, inputHash: hash }, derived: [] };
  }

  if (persisted.inputHash === hash && persisted.initializedTurn !== null) {
    const cached: CorporateSectorState = {
      schemaVersion: '10.14',
      revision: persisted.revision,
      enabled: true,
      turn: inputs.turn,
      national: {
        climate: persisted.climateBands.national || 'neutral',
        profitOutlook: persisted.outlookBands.national || 'stable',
        confidence: persisted.confidenceBands.national || 'balanced',
        industryHealthScore: 50,
        capitalAlignmentScore: 50,
        demandSupportScore: 50,
        compositeScore: 50,
        dominantConstraint: null,
        narrative: 'Corporate sector (cached bands).',
        evidence: []
      },
      regions: {},
      history: persisted.history,
      inputHash: hash,
      computeMs: 0
    };
    return { state: cached, persisted, derived: [] };
  }

  const indH = scoreIndustryHealth(inputs.industries);
  const capH = scoreCapitalAlignment(inputs.strategicCapital);
  const demH = scoreDemandSupport(inputs.householdEconomy);
  const composite = v1014Clamp(indH.score * 0.45 + capH.score * 0.3 + demH.score * 0.25, 5, 95);
  const rawClimate = v1014CorporateBandFromScore(composite, [
    { min: 78, band: 'expansionary' },
    { min: 62, band: 'supportive' },
    { min: 45, band: 'neutral' },
    { min: 28, band: 'cautious' },
    { min: 0, band: 'distressed' }
  ]);
  const climate = v1014ApplyHysteresisClimate(persisted.climateBands.national, rawClimate, composite);
  const profitOutlook = v1014OutlookFromScore(composite * 0.7 + demH.score * 0.3);
  const confidence = v1014ConfidenceFromScore((composite + capH.score) / 2);
  const national: NationalCorporateSectorSummary = {
    climate,
    profitOutlook,
    confidence,
    industryHealthScore: v1014CorporateRound(indH.score),
    capitalAlignmentScore: v1014CorporateRound(capH.score),
    demandSupportScore: v1014CorporateRound(demH.score),
    compositeScore: v1014CorporateRound(composite),
    dominantConstraint: indH.constraint,
    narrative: `Corporate sector outlook: ${climate} climate, ${profitOutlook} profit outlook, ${confidence} confidence.`,
    evidence: [...indH.evidence, ...capH.evidence, ...demH.evidence].slice(0, V1014_LIMITS.evidence)
  };
  const regions = buildRegionalOutlooks(inputs.industries, national, persisted, inputs.turn);
  const derived: CorporateSectorHistoryEntry[] = [];
  const nextPersisted: CorporateSectorPersisted = {
    ...persisted,
    revision: persisted.revision + 1,
    initializedTurn: persisted.initializedTurn ?? inputs.turn,
    lastUpdatedTurn: inputs.turn,
    inputHash: hash,
    climateBands: { ...persisted.climateBands, national: climate, ...Object.fromEntries(Object.entries(regions).map(([k, r]) => [k, r.climate])) },
    outlookBands: { ...persisted.outlookBands, national: profitOutlook, ...Object.fromEntries(Object.entries(regions).map(([k, r]) => [k, r.profitOutlook])) },
    confidenceBands: { ...persisted.confidenceBands, national: confidence, ...Object.fromEntries(Object.entries(regions).map(([k, r]) => [k, r.confidence])) },
    history: persisted.history
  };
  if (persisted.climateBands.national && persisted.climateBands.national !== climate) {
    derived.push({
      id: `ce_climate_${inputs.turn}_${climate}`,
      turn: inputs.turn,
      kind: 'climate_shifted',
      fromBand: persisted.climateBands.national,
      toBand: climate,
      summary: `National corporate climate shifted from ${persisted.climateBands.national} to ${climate}.`
    });
  }
  if (persisted.outlookBands.national && V1014_OUTLOOK_RANK[profitOutlook] < V1014_OUTLOOK_RANK[persisted.outlookBands.national]) {
    derived.push({
      id: `ce_outlook_${inputs.turn}_down`,
      turn: inputs.turn,
      kind: 'outlook_weakened',
      fromBand: persisted.outlookBands.national,
      toBand: profitOutlook,
      summary: `Firm profit outlook weakened to ${profitOutlook}.`
    });
  }
  if (persisted.outlookBands.national && V1014_OUTLOOK_RANK[profitOutlook] > V1014_OUTLOOK_RANK[persisted.outlookBands.national]) {
    derived.push({
      id: `ce_outlook_${inputs.turn}_up`,
      turn: inputs.turn,
      kind: 'outlook_strengthened',
      fromBand: persisted.outlookBands.national,
      toBand: profitOutlook,
      summary: `Firm profit outlook strengthened to ${profitOutlook}.`
    });
  }
  nextPersisted.history = [...derived, ...nextPersisted.history].slice(0, V1014_LIMITS.history);
  const computeMs = (typeof performance !== 'undefined' ? performance.now() : Date.now()) - t0;
  const state: CorporateSectorState = {
    schemaVersion: '10.14',
    revision: nextPersisted.revision,
    enabled: true,
    turn: inputs.turn,
    national,
    regions,
    history: nextPersisted.history,
    inputHash: hash,
    computeMs
  };
  return { state, persisted: nextPersisted, derived };
}

export function validateCorporateSectorState(state: CorporateSectorState | null | undefined): { valid: boolean; errors: string[] } {
  const errors: string[] = [];
  if (!state) return { valid: false, errors: ['null state'] };
  if (!V1014_CLIMATE_ORDER.includes(state.national.climate)) errors.push('invalid national climate');
  if (!V1014_OUTLOOK_ORDER.includes(state.national.profitOutlook)) errors.push('invalid profit outlook');
  if (!V1014_CONF_ORDER.includes(state.national.confidence)) errors.push('invalid confidence');
  return { valid: errors.length === 0, errors };
}

const clamp01 = (n: number) => Math.max(0, Math.min(100, n));

const bandScore = (band: string, map: Record<string, number>, fallback = 50): number =>
  typeof map[band] === 'number' ? map[band] : fallback;

  moderate: 55,
  strong: 72,

};

}

export function buildNationalEconomicAnalyticsInputs(params: {

function derivePrivateLiquidityScore(inp: NationalEconomicAnalyticsInputs): number {
  const cashScore = clamp01(Math.log10(Math.max(1, inp.spendableCash + inp.teamTreasury)) * 22);
  const worthScore = clamp01(Math.log10(Math.max(1, inp.netWorth)) * 20);
  const creditScore = clamp01(inp.creditScore);
  let phaseMod = 0;
  if (inp.economicPhase === 'recovery' || inp.economicPhase === 'survival') phaseMod = -12;
  if (inp.economicPhase === 'compounding' || inp.economicPhase === 'expansion') phaseMod = 6;
  return clamp01(Math.round(cashScore * 0.35 + worthScore * 0.25 + creditScore * 0.4 + phaseMod));
}

function deriveCompositeHealth(
  privateScore: number,
  fiscalScore: number | null,
  householdScore: number | null
): { index: number; band: EconomicHealthBand } {
  const weights: number[] = [0.35];
  const scores: number[] = [privateScore];
  if (fiscalScore != null) {
    scores.push(fiscalScore);
    weights.push(0.35);
  }
  if (householdScore != null) {
    scores.push(householdScore);
    weights.push(0.3);
  }
  const wSum = weights.reduce((a, b) => a + b, 0);
  const index = clamp01(
    Math.round(scores.reduce((acc, s, i) => acc + s * (weights[i] / wSum), 0))
  );
  const band: EconomicHealthBand =
    index >= 82 ? 'strong' :
    index >= 68 ? 'healthy' :
    index >= 50 ? 'mixed' :
    index >= 32 ? 'stressed' : 'critical';
  return { index, band };
}

function deriveBindingConstraints(
  inp: NationalEconomicAnalyticsInputs,
  privateScore: number,
  pf: NationalPublicFinanceState | null,
  hh: NationalHouseholdEconomyState | null
): EconomicBindingConstraint[] {
  const out: EconomicBindingConstraint[] = [];
  if (privateScore < 35) {
    out.push({
      id: 'bind_private_liquidity',
      system: 'private',
      severity: privateScore < 22 ? 'critical' : 'warning',
      message: 'Private liquidity and credit headroom are the binding constraint on tactical investments.'
    });
  }
  if (pf) {
    const headroom = pf.fiscalPosition.fiscalHeadroom;
    const pressure = pf.fiscalPosition.pressure;
    if (pressure === 'critical' || pressure === 'strained' || headroom < 18) {
      out.push({
        id: 'bind_fiscal_capacity',
        system: 'fiscal',
        severity: pressure === 'critical' || headroom < 10 ? 'critical' : 'warning',
        message: 'National fiscal headroom (V10.12) is limiting simultaneous public programs.'
      });
    }
  }
  if (hh) {
    const aff = hh.national.affordability;
    const housing = hh.national.housingPressure;
    if (aff === 'severe_pressure' || housing === 'severe' || housing === 'high') {
      out.push({
        id: 'bind_household_affordability',
        system: 'households',
        severity: aff === 'severe_pressure' || housing === 'severe' ? 'critical' : 'warning',
        message: 'Household affordability and housing pressure (V10.13) are absorbing growth gains.'
      });
    }
  }
  if (!out.length) {
    out.push({
      id: 'bind_balanced',
      system: 'synthesis',
      severity: 'info',
      message: 'No single economic layer is clearly binding; monitor fiscal and household drift.'
    });
  }
  out.sort((a, b) => {
    const rank = { critical: 0, warning: 1, info: 2 };
    return rank[a.severity] - rank[b.severity];
  });
  return out.slice(0, 4);
}

export function computeNationalEconomicAnalytics(
  inputs: NationalEconomicAnalyticsInputs
): NationalEconomicAnalyticsSnapshot {
  const inputHash = nationalEconomicAnalyticsInputHash(inputs);
  if (!inputs.enabled) {
    return {
      schemaVersion: V1014_SCHEMA_VERSION,
      revision: 1,
      enabled: false,
      turn: inputs.turn,
      day: inputs.day,
      compositeHealthIndex: 0,
      healthBand: 'mixed',
      privateEconomy: {
        liquidityScore: 0,
        creditBand: 'n/a',
        economicPhase: inputs.economicPhase,
        headline: 'National economic analytics require National Systems.'
      },
      publicEconomy: null,
      householdEconomy: null,
      bindingConstraints: [],
      kpis: [],
      headline: 'Economic analytics paused (National Systems off).',
      inputHash
    };
  }

  const privateScore = derivePrivateLiquidityScore(inputs);
  const creditBand = getCreditScoreRange(inputs.creditScore).label;
  const pf = inputs.publicFinance;
  const hh = inputs.households;

  const fiscalScore = pf
    ? clamp01(
        Math.round(
          FISCAL_HEADROOM_SCORE[pf.fiscalPosition.headroomBand] * 0.55 +
            FISCAL_PRESSURE_SCORE[pf.fiscalPosition.pressure] * 0.45
        )
      )
    : null;

  const householdScore = hh
    ? clamp01(
        Math.round(
          bandScore(hh.national.prosperity, PROSPERITY_SCORE) * 0.45 +
            bandScore(hh.national.affordability, AFFORDABILITY_SCORE) * 0.55
        )
      )
    : null;

  const { index, band } = deriveCompositeHealth(privateScore, fiscalScore, householdScore);
  const bindingConstraints = deriveBindingConstraints(inputs, privateScore, pf, hh);

  const kpis: EconomicAnalyticsKpi[] = [
    {
      id: 'composite',
      label: 'Composite economic health',
      valueText: `${index}/100`,
      band
    },
    {
      id: 'private_liquidity',
      label: 'Private liquidity',
      valueText: `${privateScore}/100`,
      band: creditBand
    }
  ];
  if (pf) {
    kpis.push({
      id: 'fiscal_headroom',
      label: 'Fiscal headroom (V10.12)',
      valueText: `${Math.round(pf.fiscalPosition.fiscalHeadroom)} pts`,
      band: pf.fiscalPosition.headroomBand,
      trend: pf.revenue.trend === 'rising' ? 'up' : pf.revenue.trend === 'weakening' ? 'down' : 'flat'
    });
  }
  if (hh) {
    kpis.push({
      id: 'living_standards',
      label: 'Living standards (V10.13)',
      valueText: `${hh.national.prosperity} / ${hh.national.affordability}`,
      band: hh.national.trajectory,
      trend:
        hh.national.trajectory === 'improving' ? 'up' :
        hh.national.trajectory === 'weakening' || hh.national.trajectory === 'deteriorating' ? 'down' : 'flat'
    });
  }

  const top = bindingConstraints[0];
  const headline = top
    ? `Day ${inputs.day}: ${top.message}`
    : `Day ${inputs.day}: Balanced economic posture (${band}).`;

  return {
    schemaVersion: V1014_SCHEMA_VERSION,
    revision: 1,
    enabled: true,
    turn: inputs.turn,
    day: inputs.day,
    compositeHealthIndex: index,
    healthBand: band,
    privateEconomy: {
      liquidityScore: privateScore,
      creditBand,
      economicPhase: inputs.economicPhase,
      headline: `Spendable $${Math.round(inputs.spendableCash).toLocaleString()} · phase ${inputs.economicPhase}`
    },
    publicEconomy: pf
      ? {
          fiscalHeadroom: pf.fiscalPosition.fiscalHeadroom,
          headroomBand: pf.fiscalPosition.headroomBand,
          pressure: pf.fiscalPosition.pressure,
          sustainability: pf.fiscalPosition.sustainability,
          revenueTrend: pf.revenue.trend
        }
      : null,
    householdEconomy: hh
      ? {
          prosperity: hh.national.prosperity,
          affordability: hh.national.affordability,
          trajectory: hh.national.trajectory,
          domesticDemand: hh.national.domesticDemand.level
        }
      : null,
    bindingConstraints,
    kpis,
    headline,
    inputHash
  };
}

export function validateNationalEconomicAnalyticsSnapshot(
  snap: NationalEconomicAnalyticsSnapshot | null | undefined
): { valid: boolean; issues: string[] } {
  const issues: string[] = [];
  if (!snap) return { valid: false, issues: ['snapshot is null'] };
  if (snap.schemaVersion !== V1014_SCHEMA_VERSION) issues.push('unexpected schemaVersion');
  if (!Number.isFinite(snap.compositeHealthIndex)) issues.push('non-finite compositeHealthIndex');
  if (snap.enabled && snap.kpis.length < 2) issues.push('expected at least private KPIs when enabled');
  return { valid: issues.length === 0, issues };
}

export function summarizeNationalEconomicAnalyticsForPlay(
  snap: NationalEconomicAnalyticsSnapshot | null | undefined
): string {
  if (!snap || !snap.enabled) return 'Economic analytics unavailable.';
  return snap.headline;
}

function v1014ScoreToAgeBand(score: number): AgeStructureBand {
  if (score >= 78) return 'youthful';
  if (score >= 62) return 'balanced';
  if (score >= 46) return 'mature';
  if (score >= 30) return 'aging';
  return 'super_aging';
}

function v1014ScoreToMigrationBand(score: number): MigrationIntentBand {
  if (score >= 72) return 'strong_inward';
  if (score >= 55) return 'inward';
  if (score >= 42) return 'neutral';
  return 'outward';
}

function v1014ScoreToVitalityBand(score: number): DemographicVitalityBand {
  if (score >= 72) return 'dynamic';
  if (score >= 56) return 'growing';
  if (score >= 40) return 'stable';
  return 'declining';
}

function v1014ScoreToTrajectoryBand(score: number): PopulationTrajectoryBand {
  if (score >= 68) return 'improving';
  if (score >= 48) return 'stable';
  if (score >= 32) return 'weakening';
  return 'stressed';
}

export function sanitizePopulationDynamicsPersisted(raw: unknown, turn = 1): PopulationDynamicsPersisted {
  const base = createInitialPopulationDynamicsPersisted(turn);
  if (!raw || typeof raw !== 'object') return base;
  const r = raw as Partial<PopulationDynamicsPersisted>;
  return {
    ...base,
    schemaVersion: typeof r.schemaVersion === 'string' ? r.schemaVersion : V1014_SCHEMA_VERSION,
    version: typeof r.version === 'string' ? r.version : V1014_SCHEMA_VERSION,
    revision: Math.max(0, Math.floor(Number(r.revision) || 0)),
    lastComputedTurn: Math.max(1, Math.floor(Number(r.lastComputedTurn) || turn)),
    lastInputHash: typeof r.lastInputHash === 'string' ? r.lastInputHash : '',
    pinnedNationalVitality: r.pinnedNationalVitality,
    pinnedNationalMigration: r.pinnedNationalMigration,
    history: Array.isArray(r.history)
      ? r.history.slice(-V1014_LIMITS.maxHistory).map((h: any) => ({
          turn: Math.max(1, Math.floor(Number(h?.turn) || 1)),
          nationalVitality: (h?.nationalVitality as DemographicVitalityBand) || 'stable',
          nationalMigration: (h?.nationalMigration as MigrationIntentBand) || 'neutral',
          nationalTrajectory: (h?.nationalTrajectory as PopulationTrajectoryBand) || 'stable',
          summary: String(h?.summary || '')
        }))
      : [],
    transitions: Array.isArray(r.transitions)
      ? r.transitions.slice(-V1014_LIMITS.maxTransitions).map((t: any, i: number) => ({
          id: String(t?.id || `pd_tr_${i}`),
          turn: Math.max(1, Math.floor(Number(t?.turn) || 1)),
          regionId: t?.regionId === 'national' ? 'national' : String(t?.regionId || 'NSW'),
          kind: (t?.kind as PopulationDynamicsTransition['kind']) || 'vitality',
          fromBand: String(t?.fromBand || ''),
          toBand: String(t?.toBand || ''),
          reason: String(t?.reason || '')
        }))
      : []
  };
}

export function populationDynamicsInputHash(inputs: PopulationDynamicsInputs): string {
  const keys = Object.keys(inputs.regions || {}).sort();
  const parts = keys.map((k) => {
    const r = inputs.regions[k];
    return [
      k,
      r.lrDevelopment ?? 0,
      r.lrMomentum ?? 0,
      r.housingPressure ?? '',
      r.employmentOpportunity ?? '',
      r.prosperity ?? '',
      r.workforcePressure ?? ''
    ].join(':');
  });
  return `v1014|${inputs.turn}|${inputs.enabled ? 1 : 0}|${parts.join('|')}`;
}

export function buildPopulationDynamicsInputs(params: {
  turn: number;
  gameSettings?: any;
  livingRegions?: any;
  workforce?: NationalWorkforceState | null;
  households?: NationalHouseholdEconomyState | null;
}): PopulationDynamicsInputs {
  const gset = params.gameSettings || {};
  const enabled = gset.populationDynamicsEnabled !== false && gset.nationalSystemsEnabled !== false;
  const lr = params.livingRegions;
  const wf = params.workforce;
  const hh = params.households;
  const regions: Record<string, PopulationDynamicsRegionInputs> = {};
  for (const regionId of V1014_CORE_REGIONS) {
    const lrReg = lr?.regions?.[regionId];
    const hhReg = hh?.regions?.[regionId];
    const wfReg = wf?.regions?.[regionId];
    regions[regionId] = {
      regionId,
      lrDevelopment: lrReg ? Number(lrReg.developmentScore ?? lrReg.development ?? 50) : 50,
      lrMomentum: lrReg ? Number(lrReg.momentumScore ?? lrReg.momentum ?? 50) : 50,
      housingPressure: hhReg?.housingPressure,
      employmentOpportunity: hhReg?.employmentOpportunity,
      prosperity: hhReg?.prosperity,
      workforcePressure: wfReg?.pressureBand
    };
  }
  return {
    turn: Math.max(1, Math.floor(Number(params.turn) || 1)),
    enabled,
    regions
  };
}

function v1014ApplyHysteresisBand<T extends string>(
  rawBand: T,
  pinned: T | undefined,
  rankMap: Record<string, number>,
  deadband: number,
  rawScore: number,
  pinScore: number
): T {
  if (!pinned) return rawBand;
  const delta = rawScore - pinScore;
  if (Math.abs(delta) < deadband) return pinned;
  return rawBand;
}

function computeRegionPopulationDynamics(inp: PopulationDynamicsRegionInputs): PopulationDynamicsRegionState {
  const dev = v1014Clamp(inp.lrDevelopment ?? 50, 0, 100);
  const mom = v1014Clamp(inp.lrMomentum ?? 50, 0, 100);
  const empRank = inp.employmentOpportunity ? V1013_EMPLOYMENT_RANK[inp.employmentOpportunity] : 2;
  const houseRank = inp.housingPressure ? V1014_VITALITY_RANK[
    inp.housingPressure === 'severe' || inp.housingPressure === 'high' ? 'declining' : 'stable'
  ] : 1;
  const wfRank = inp.workforcePressure
    ? WORKFORCE_PRESSURE_BANDS.indexOf(inp.workforcePressure)
    : 2;

  const vitalityScore = v1014Clamp(dev * 0.35 + mom * 0.35 + empRank * 12 + (3 - houseRank) * 8, 0, 100);
  const migrationScore = v1014Clamp(
    empRank * 18 + (inp.prosperity ? V1013_PROSPERITY_RANK[inp.prosperity] * 10 : 20) - (inp.housingPressure === 'severe' ? 22 : inp.housingPressure === 'high' ? 12 : 0),
    0,
    100
  );
  const ageScore = v1014Clamp(72 - wfRank * 9 + (inp.prosperity ? V1013_PROSPERITY_RANK[inp.prosperity] * 4 : 0), 0, 100);
  const trajectoryScore = v1014Clamp(vitalityScore * 0.45 + migrationScore * 0.35 + mom * 0.2, 0, 100);

  const vitality = v1014ScoreToVitalityBand(vitalityScore);
  const migrationIntent = v1014ScoreToMigrationBand(migrationScore);
  const ageStructure = v1014ScoreToAgeBand(ageScore);
  const trajectory = v1014ScoreToTrajectoryBand(trajectoryScore);

  return {
    regionId: inp.regionId,
    ageStructure,
    migrationIntent,
    vitality,
    trajectory,
    summary: `${inp.regionId}: ${V1014_VITALITY_LABELS[vitality]} vitality, ${V1014_MIGRATION_LABELS[migrationIntent]} migration intent, ${V1014_AGE_LABELS[ageStructure]} age profile.`
  };
}

export function computeNationalPopulationDynamics(
  inputs: PopulationDynamicsInputs,
  persistedIn: PopulationDynamicsPersisted,
  prevState: NationalPopulationDynamicsState | null
): { state: NationalPopulationDynamicsState; persisted: PopulationDynamicsPersisted } {
  const persisted = sanitizePopulationDynamicsPersisted(persistedIn, inputs.turn);
  const inputHash = populationDynamicsInputHash(inputs);
  if (!inputs.enabled) {
    const off: NationalPopulationDynamicsState = {
      schemaVersion: V1014_SCHEMA_VERSION,
      revision: persisted.revision,
      lastUpdatedTurn: inputs.turn,
      lastInputHash: inputHash,
      enabled: false,
      national: {
        ageStructure: 'balanced',
        migrationIntent: 'neutral',
        vitality: 'stable',
        trajectory: 'stable'
      },
      regions: {},
      transitions: persisted.transitions,
      history: persisted.history
    };
    return { state: off, persisted: { ...persisted, lastInputHash: inputHash, lastComputedTurn: inputs.turn } };
  }

  if (persisted.lastInputHash === inputHash && prevState && prevState.lastInputHash === inputHash) {
    return { state: prevState, persisted };
  }

  const regionStates: Record<string, PopulationDynamicsRegionState> = {};
  const regionKeys = Object.keys(inputs.regions).length ? Object.keys(inputs.regions) : [...V1014_CORE_REGIONS];
  for (const rid of regionKeys) {
    const inp = inputs.regions[rid] || { regionId: rid };
    regionStates[rid] = computeRegionPopulationDynamics({ ...inp, regionId: rid });
  }

  const avg = (fn: (r: PopulationDynamicsRegionState) => number) => {
    const vals = Object.values(regionStates).map(fn);
    return vals.length ? vals.reduce((a, b) => a + b, 0) / vals.length : 1;
  };

  const rawVitality = v1014ScoreToVitalityBand(avg((r) => V1014_VITALITY_RANK[r.vitality] * 25 + 12));
  const rawMigration = v1014ScoreToMigrationBand(avg((r) => V1014_MIGRATION_RANK[r.migrationIntent] * 25 + 12));
  const rawAge = v1014ScoreToAgeBand(avg((r) => (4 - V1014_AGE_RANK[r.ageStructure]) * 22 + 20));
  const rawTrajectory = v1014ScoreToTrajectoryBand(avg((r) => {
    const t = r.trajectory;
    return t === 'improving' ? 75 : t === 'stable' ? 55 : t === 'weakening' ? 38 : 22;
  }));

  const vitality = v1014ApplyHysteresisBand(
    rawVitality,
    persisted.pinnedNationalVitality,
    V1014_VITALITY_RANK,
    V1014_LIMITS.hysteresisDeadband,
    V1014_VITALITY_RANK[rawVitality],
    persisted.pinnedNationalVitality ? V1014_VITALITY_RANK[persisted.pinnedNationalVitality] : V1014_VITALITY_RANK[rawVitality]
  );
  const migrationIntent = v1014ApplyHysteresisBand(
    rawMigration,
    persisted.pinnedNationalMigration,
    V1014_MIGRATION_RANK,
    V1014_LIMITS.hysteresisDeadband,
    V1014_MIGRATION_RANK[rawMigration],
    persisted.pinnedNationalMigration ? V1014_MIGRATION_RANK[persisted.pinnedNationalMigration] : V1014_MIGRATION_RANK[rawMigration]
  );

  const transitions = [...persisted.transitions];
  const pushTransition = (regionId: string | 'national', kind: PopulationDynamicsTransition['kind'], fromBand: string, toBand: string, reason: string) => {
    if (fromBand === toBand) return;
    transitions.push({
      id: `pd_${kind}_${inputs.turn}_${regionId}_${transitions.length}`,
      turn: inputs.turn,
      regionId,
      kind,
      fromBand,
      toBand,
      reason
    });
  };

  if (prevState?.national.vitality && prevState.national.vitality !== vitality) {
    pushTransition('national', 'vitality', prevState.national.vitality, vitality, 'National vitality band shift from regional conditions.');
  }
  if (prevState?.national.migrationIntent && prevState.national.migrationIntent !== migrationIntent) {
    pushTransition('national', 'migration', prevState.national.migrationIntent, migrationIntent, 'National migration intent shift from employment and affordability signals.');
  }

  const history = [
    ...persisted.history,
    {
      turn: inputs.turn,
      nationalVitality: vitality,
      nationalMigration: migrationIntent,
      nationalTrajectory: rawTrajectory,
      summary: `National demographic momentum: ${V1014_VITALITY_LABELS[vitality]}, ${V1014_MIGRATION_LABELS[migrationIntent]}.`
    }
  ].slice(-V1014_LIMITS.maxHistory);

  const nextPersisted: PopulationDynamicsPersisted = {
    ...persisted,
    revision: persisted.revision + 1,
    lastComputedTurn: inputs.turn,
    lastInputHash: inputHash,
    pinnedNationalVitality: vitality,
    pinnedNationalMigration: migrationIntent,
    history,
    transitions: transitions.slice(-V1014_LIMITS.maxTransitions)
  };

  const state: NationalPopulationDynamicsState = {
    schemaVersion: V1014_SCHEMA_VERSION,
    revision: nextPersisted.revision,
    lastUpdatedTurn: inputs.turn,
    lastInputHash: inputHash,
    enabled: true,
    national: {
      ageStructure: rawAge,
      migrationIntent,
      vitality,
      trajectory: rawTrajectory
    },
    regions: regionStates,
    transitions: nextPersisted.transitions,
    history: nextPersisted.history
  };

  return { state, persisted: nextPersisted };
}

export interface V1014MobilityContext {
  regionId: string;
  boundedModifier: number;
  relocatesPopulation: false;
  explanation: string;
}

export function computePopulationMobilityModifier(
  state: NationalPopulationDynamicsState | null | undefined,
  regionId: string
): V1014MobilityContext {
  const reg = state?.regions?.[regionId];
  if (!reg || !state?.enabled) {
    return {
      regionId,
      boundedModifier: 0,
      relocatesPopulation: false,
      explanation: `Neutral demographic mobility modifier for ${regionId}.`
    };
  }
  let mod = 0;
  if (V1014_MIGRATION_RANK[reg.migrationIntent] >= 2) mod += 0.35;
  if (V1014_VITALITY_RANK[reg.vitality] >= 2) mod += 0.25;
  if (reg.trajectory === 'stressed') mod -= 0.45;
  else if (reg.trajectory === 'weakening') mod -= 0.2;
  if (reg.ageStructure === 'super_aging') mod -= 0.15;
  const boundedModifier = v1014Clamp(Number(mod.toFixed(2)), -1, 1);
  return {
    regionId,
    boundedModifier,
    relocatesPopulation: false,
    explanation: `${regionId} demographic momentum (${V1014_VITALITY_LABELS[reg.vitality]}, ${V1014_MIGRATION_LABELS[reg.migrationIntent]}) yields bounded modifier ${boundedModifier >= 0 ? `+${boundedModifier}` : boundedModifier} (no physical population relocation).`
  };
}

export function populationDynamicsActionAlignment(
  state: NationalPopulationDynamicsState | null | undefined,
  action: { id?: string; title?: string; regionId?: string | null }
): { relevanceDelta: number; reason: string | null } {
  if (!state?.enabled || !action?.regionId) return { relevanceDelta: 0, reason: null };
  const reg = state.regions[action.regionId];
  if (!reg) return { relevanceDelta: 0, reason: null };
  if (reg.trajectory === 'stressed' && /housing|urban|settlement/i.test(String(action.title || action.id || ''))) {
    return { relevanceDelta: 1.5, reason: 'eases demographic stress in a strained region' };
  }
  if (V1014_MIGRATION_RANK[reg.migrationIntent] >= 2 && /skills|training|education/i.test(String(action.title || action.id || ''))) {
    return { relevanceDelta: 1.25, reason: 'supports inward talent momentum' };
  }
  return { relevanceDelta: 0, reason: null };
}

  };
}

export function runV1014PopulationDynamicsSelfTests(): {
  passed: number;
  failed: number;
  total: number;
  results: Array<{ id: string; passed: boolean; detail: string }>;
} {
  const results: Array<{ id: string; passed: boolean; detail: string }> = [];
  const check = (id: string, passed: boolean, detail = '') => results.push({ id, passed, detail: detail || (passed ? 'ok' : 'fail') });

  const init = createInitialPopulationDynamicsPersisted(1);
  const baseInp = makeDefaultV1014TestInputs();
  const base = computeNationalPopulationDynamics(baseInp, init, null);
  check('pd01', base.state.enabled && base.state.revision === 1, `revision=${base.state.revision}`);
  check('pd02', Object.keys(base.state.regions).length === V1014_CORE_REGIONS.length);
  check('pd03', base.state.national.vitality !== 'declining' || base.state.national.migrationIntent !== 'outward');

  const off = computeNationalPopulationDynamics({ ...baseInp, enabled: false }, init, null);
  check('pd04', !off.state.enabled && off.state.regions && Object.keys(off.state.regions).length === 0);

  const detA = computeNationalPopulationDynamics(baseInp, init, null);
  const detB = computeNationalPopulationDynamics(baseInp, init, null);
  check('pd05', JSON.stringify(detA.state) === JSON.stringify(detB.state));

  const vicMod = computePopulationMobilityModifier(base.state, 'VIC');
  check('pd06', vicMod.relocatesPopulation === false && vicMod.boundedModifier >= -1 && vicMod.boundedModifier <= 1);

  const nswMod = computePopulationMobilityModifier(base.state, 'NSW');
  check('pd07', nswMod.relocatesPopulation === false);

  const forbiddenKeys = ['population', 'unemploymentRate', 'workerAgents', 'payrollCost', 'wageBill', 'migrationQuota', 'populationCount'];
  check(
    'pd08',
    forbiddenKeys.every((k) => !(k in base.state) && !(k in base.state.national)),
    'forbidden literal demographic keys absent'
  );

  const restored = sanitizePopulationDynamicsPersisted(JSON.parse(JSON.stringify(base.persisted)));
  check('pd09', restored.lastInputHash === base.persisted.lastInputHash);

  const legacy = sanitizePopulationDynamicsPersisted(undefined);
  check('pd10', legacy.schemaVersion === V1014_SCHEMA_VERSION);

  const hashStable = populationDynamicsInputHash(baseInp) === populationDynamicsInputHash(baseInp);
  check('pd11', hashStable);

  const built = buildPopulationDynamicsInputs({
    turn: 3,
    gameSettings: { nationalSystemsEnabled: true, populationDynamicsEnabled: true },
    livingRegions: null,
    workforce: null,
    households: null
  });
  check('pd12', built.enabled && Object.keys(built.regions).length === V1014_CORE_REGIONS.length);

  const align = populationDynamicsActionAlignment(base.state, { title: 'Expand urban housing program', regionId: 'VIC' });
  check('pd13', align.relevanceDelta >= 0);

  let pers = base.persisted;
  let prev: NationalPopulationDynamicsState | null = base.state;
  for (let i = 0; i < 20; i++) {
    const inp = makeDefaultV1014TestInputs({ turn: 5 + i });
    const r = computeNationalPopulationDynamics(inp, pers, prev);
    pers = r.persisted;
    prev = r.state;
  }
  check('pd14', (pers.history.length || 0) <= V1014_LIMITS.maxHistory);

  const startMs = Date.now();
  pers = createInitialPopulationDynamicsPersisted();
  prev = null;
  for (let i = 0; i < 25; i++) {
    const r = computeNationalPopulationDynamics(makeDefaultV1014TestInputs({ turn: i + 1 }), pers, prev);
    pers = r.persisted;
    prev = r.state;
  }
  check('pd15', Date.now() - startMs < 250, `perfMs=${Date.now() - startMs}`);

  const passed = results.filter((r) => r.passed).length;
  return { passed, failed: results.length - passed, total: results.length, results };
}

function bandFromScore(score: number, prev?: ServiceDeliveryBand, hasPrev?: boolean): ServiceDeliveryBand {
  const raw: ServiceDeliveryBand =
    score >= 82 ? 'excellent' : score >= 68 ? 'strong' : score >= 52 ? 'adequate' : score >= 36 ? 'strained' : 'critical';
  if (!hasPrev || !prev || prev === raw) return raw;
  const db = V1014_LIMITS.hysteresisDeadband;
  if (prev === 'strong' && raw === 'adequate' && score >= 68 - db) return 'strong';
  if (prev === 'adequate' && raw === 'strained' && score >= 52 - db) return 'adequate';
  return raw;
}

export function createInitialPublicServicesPersisted(turn = 1): PublicServicesPersisted {
  const base: ServiceDeliveryBand = 'adequate';
  const domainBands = Object.fromEntries(PUBLIC_SERVICE_DOMAINS.map(d => [d, base])) as Record<
    PublicServiceDomain,
    ServiceDeliveryBand
  >;
  return {
    schemaVersion: V1014_SCHEMA_VERSION,
    revision: 0,
    lastInputHash: '',
    lastTurn: Math.max(1, turn),
    domainBands,
    nationalOverallBand: base,
    regionalBandMemory: {},
    transitions: [],
    history: []
  };
}

export function sanitizePublicServicesPersisted(raw: unknown, turn = 1): PublicServicesPersisted | null {
  if (!raw || typeof raw !== 'object') return null;
  const r = raw as Partial<PublicServicesPersisted>;
  const init = createInitialPublicServicesPersisted(turn);
  const domainBands = { ...init.domainBands };
  if (r.domainBands && typeof r.domainBands === 'object') {
    for (const d of PUBLIC_SERVICE_DOMAINS) {
      const b = (r.domainBands as Record<string, string>)[d];
      if (SERVICE_DELIVERY_BANDS.includes(b as ServiceDeliveryBand)) domainBands[d] = b as ServiceDeliveryBand;
    }
  }
  const regionalBandMemory: Record<string, ServiceDeliveryBand> = {};
  if (r.regionalBandMemory && typeof r.regionalBandMemory === 'object') {
    for (const [k, v] of Object.entries(r.regionalBandMemory)) {
      if (V1014_CORE_REGIONS.includes(k) && SERVICE_DELIVERY_BANDS.includes(v as ServiceDeliveryBand)) {
        regionalBandMemory[k] = v as ServiceDeliveryBand;
      }
    }
  }
  return {
    schemaVersion: String(r.schemaVersion || V1014_SCHEMA_VERSION),
    revision: Math.max(0, Math.floor(Number(r.revision) || 0)),
    lastInputHash: String(r.lastInputHash || ''),
    lastTurn: Math.max(1, Math.floor(Number(r.lastTurn) || turn)),
    domainBands,
    nationalOverallBand: SERVICE_DELIVERY_BANDS.includes(r.nationalOverallBand as ServiceDeliveryBand)
      ? (r.nationalOverallBand as ServiceDeliveryBand)
      : init.nationalOverallBand,
    regionalBandMemory,
    transitions: Array.isArray(r.transitions) ? r.transitions.slice(-V1014_LIMITS.maxTransitions) : [],
    history: Array.isArray(r.history) ? r.history.slice(-V1014_LIMITS.maxHistory) : []
  };
}

function stableJson(obj: unknown): string {
  try {
    return JSON.stringify(obj);
  } catch {
    return '';
  }
}

export function publicServicesInputHash(inputs: PublicServicesSystemInputs): string {
  return stableJson({
    turn: inputs.turn,
    enabled: inputs.enabled,
    sc: inputs.stateCapacity?.national?.deliveryReadiness ?? null,
    pf: inputs.publicFinance?.national?.fiscalHeadroomBand ?? null,
    hh: inputs.households?.national?.essentialAccess ?? null,
    stab: inputs.publicStabilityScore ?? null,
    infra: inputs.completedInfrastructureSupport ?? null,
    crises: inputs.activeCrises ?? 0
  });
}

export function buildPublicServicesInputs(params: {
  turn: number;
  gameState?: any;
  gameSettings?: any;
  stateCapacity?: unknown;
  publicFinance?: unknown;
  households?: unknown;
}): PublicServicesSystemInputs {
  const gs = params.gameState || {};
  const gset = params.gameSettings || {};
  const turn = Math.max(1, Math.floor(Number(params.turn) || 1));
  const ps = gs.publicStabilityState;
  const stability =
    typeof ps === 'number' ? ps : ps && typeof ps.nationalStability === 'number' ? Number(ps.nationalStability) : null;
  const crises = Array.isArray(gs.crisisChainState?.activeCrisisChains)
    ? gs.crisisChainState.activeCrisisChains.filter((c: any) => c && c.status !== 'resolved').length
    : 0;
  return {
    turn,
    enabled: gset.publicServicesEnabled !== false && gset.nationalSystemsEnabled !== false,
    regions: gs.regions,
    livingRegions: gs.livingRegions,
    stateCapacity: (params.stateCapacity || gs.stateCapacityState || gs.stateCapacity) as PublicServicesSystemInputs['stateCapacity'],
    publicFinance: (params.publicFinance || gs.publicFinanceState || gs.publicFinance) as PublicServicesSystemInputs['publicFinance'],
    households: (params.households || gs.householdEconomyState) as PublicServicesSystemInputs['households'],
    publicStabilityScore: stability,
    completedInfrastructureSupport: undefined,
    activeCrises: crises
  };
}

function domainScore(
  domain: PublicServiceDomain,
  inputs: PublicServicesSystemInputs,
  regionalBias = 0
): number {
  let score = 58 + regionalBias;
  const readiness = Number(inputs.stateCapacity?.national?.deliveryReadiness ?? 62);
  const headroom = inputs.publicFinance?.national?.fiscalHeadroomBand;
  const essential = inputs.households?.national?.essentialAccess;
  const stab = inputs.publicStabilityScore ?? 70;
  const infra = inputs.completedInfrastructureSupport ?? 65;
  const crises = inputs.activeCrises ?? 0;

  score += (readiness - 60) * 0.35;
  score += (infra - 60) * 0.15;
  score += (stab - 70) * 0.12;
  if (headroom === 'strained' || headroom === 'critical') score -= 6;
  if (headroom === 'strong' || headroom === 'comfortable') score += 4;
  if (essential === 'critical' || essential === 'poor') score -= domain === 'health' || domain === 'emergency_response' ? 10 : 4;
  if (essential === 'strong' || essential === 'excellent') score += 4;
  score -= Math.min(18, crises * 4);

  if (domain === 'education') score += 2;
  if (domain === 'emergency_response') score -= crises * 2;
  if (domain === 'justice_administration') score += (readiness - 60) * 0.08;
  if (domain === 'local_civic') score += (infra - 60) * 0.12;

  return v1014Clamp(score);
}

function pushTransition(
  list: PublicServicesTransition[],
  turn: number,
  domain: PublicServiceDomain | 'national',
  fromBand: ServiceDeliveryBand,
  toBand: ServiceDeliveryBand,
  regionId: string | null,
  reason: string
): void {
  if (fromBand === toBand) return;
  list.push({
    id: `ps_${domain}_${regionId || 'national'}_t${turn}_${list.length}`,
    turn,
    domain,
    fromBand,
    toBand,
    regionId,
    reason
  });
}

export function computeNationalPublicServices(
  inputs: PublicServicesSystemInputs,
  persistedRaw?: PublicServicesPersisted | null,
  previousState?: NationalPublicServicesState | null
): { state: NationalPublicServicesState; persisted: PublicServicesPersisted; events: PublicServicesTransition[] } {
  const turn = Math.max(1, Math.floor(Number(inputs.turn) || 1));
  const persisted = sanitizePublicServicesPersisted(persistedRaw, turn) || createInitialPublicServicesPersisted(turn);
  const hash = publicServicesInputHash(inputs);
  const events: PublicServicesTransition[] = [];

  if (!inputs.enabled) {
    const off: NationalPublicServicesState = {
      schemaVersion: V1014_SCHEMA_VERSION,
      revision: persisted.revision,
      turn,
      enabled: false,
      inputHash: hash,
      national: {
        overall: 'adequate',
        civicAccess: 'adequate',
        accessScore: 0,
        backlogPressure: 0,
        domains: { ...persisted.domainBands },
        strainedRegions: [],
        leadingRegions: [],
        summary: 'Public services module disabled.'
      },
      regions: [],
      transitions: [],
      warnings: []
    };
    return {
      state: off,
      persisted: { ...persisted, lastInputHash: hash, lastTurn: turn },
      events
    };
  }

  const domainBands = { ...persisted.domainBands };
  const newTransitions: PublicServicesTransition[] = [];
  for (const d of PUBLIC_SERVICE_DOMAINS) {
    const prev = domainBands[d];
    const score = domainScore(d, inputs);
    const next = bandFromScore(score, prev, persisted.revision > 0);
    domainBands[d] = next;
    pushTransition(newTransitions, turn, d, prev, next, null, `National ${PUBLIC_SERVICE_DOMAIN_LABEL[d]} score ${Math.round(score)}`);
  }

  const domainScores = PUBLIC_SERVICE_DOMAINS.map(d => domainScore(d, inputs));
  const nationalScore = v1014Clamp(domainScores.reduce((a, b) => a + b, 0) / domainScores.length);
  const prevNational = persisted.nationalOverallBand || previousState?.national.overall || 'adequate';
  const nationalOverall = bandFromScore(nationalScore, prevNational, persisted.revision > 0);
  pushTransition(newTransitions, turn, 'national', prevNational, nationalOverall, null, `Composite civic delivery ${Math.round(nationalScore)}`);

  const regions: PublicServicesRegionalRow[] = [];
  const strainedRegions: string[] = [];
  const leadingRegions: string[] = [];
  for (let i = 0; i < V1014_CORE_REGIONS.length; i++) {
    const regionId = V1014_CORE_REGIONS[i];
    const bias = (i % 3) * 2 - 2;
    const domMap = Object.fromEntries(
      PUBLIC_SERVICE_DOMAINS.map(d => {
        const s = domainScore(d, inputs, bias);
        const prevR = persisted.regionalBandMemory[regionId];
        const band = bandFromScore(s, prevR, Boolean(prevR));
        return [d, band];
      })
    ) as Record<PublicServiceDomain, ServiceDeliveryBand>;
    const avg = v1014Clamp(
      PUBLIC_SERVICE_DOMAINS.reduce((acc, d) => acc + SERVICE_DELIVERY_RANK[domMap[d]], 0) *
        (100 / (PUBLIC_SERVICE_DOMAINS.length * (SERVICE_DELIVERY_RANK.excellent || 4)))
    );
    const overall = bandFromScore(avg, persisted.regionalBandMemory[regionId], Boolean(persisted.regionalBandMemory[regionId]));
    if (SERVICE_DELIVERY_RANK[overall] <= 1) strainedRegions.push(regionId);
    if (SERVICE_DELIVERY_RANK[overall] >= 3) leadingRegions.push(regionId);
    regions.push({
      regionId,
      overall,
      domains: domMap,
      accessScore: Math.round(avg),
      backlogPressure: v1014Clamp(100 - avg + (inputs.activeCrises || 0) * 5),
      explanation: `${regionId}: ${SERVICE_DELIVERY_LABEL[overall]} public service delivery.`
    });
  }

  const backlogPressure = v1014Clamp(
    100 - nationalScore + Math.max(0, (inputs.stateCapacity?.national?.coordinationLoad ?? 40) - 55)
  );

  const national: PublicServicesNationalSnapshot = {
    overall: nationalOverall,
    civicAccess: nationalOverall,
    accessScore: Math.round(nationalScore),
    backlogPressure: Math.round(backlogPressure),
    domains: domainBands,
    strainedRegions: strainedRegions.slice(0, 6),
    leadingRegions: leadingRegions.slice(0, 6),
    summary:
      nationalOverall === 'excellent' || nationalOverall === 'strong'
        ? 'Core public services are keeping pace with national demand.'
        : nationalOverall === 'adequate'
        ? 'Public services are functional but uneven across regions and domains.'
        : 'Public service delivery is under pressure — backlog and access gaps are widening.'
  };

  const warnings: string[] = [];
  if (backlogPressure >= 70) warnings.push('National service backlog pressure is elevated.');
  if (strainedRegions.length >= 3) warnings.push('Multiple regions show strained civic delivery simultaneously.');

  const nextPersisted: PublicServicesPersisted = {
    ...persisted,
    schemaVersion: V1014_SCHEMA_VERSION,
    revision: persisted.revision + 1,
    lastInputHash: hash,
    lastTurn: turn,
    domainBands,
    nationalOverallBand: nationalOverall,
    regionalBandMemory: Object.fromEntries(regions.map(r => [r.regionId, r.overall])),
    transitions: [...persisted.transitions, ...newTransitions].slice(-V1014_LIMITS.maxTransitions),
    history: [...persisted.history, { turn, overall: nationalOverall, accessScore: Math.round(nationalScore) }].slice(
      -V1014_LIMITS.maxHistory
    )
  };

  const state: NationalPublicServicesState = {
    schemaVersion: V1014_SCHEMA_VERSION,
    revision: nextPersisted.revision,
    turn,
    enabled: true,
    inputHash: hash,
    national,
    regions,
    transitions: newTransitions,
    warnings
  };

  events.push(...newTransitions);
  return { state, persisted: nextPersisted, events };
}

export function validatePublicServicesState(state: NationalPublicServicesState | null): { valid: boolean; errors: string[] } {
  const errors: string[] = [];
  if (!state) return { valid: false, errors: ['null state'] };
  if (state.schemaVersion !== V1014_SCHEMA_VERSION) errors.push(`schema ${state.schemaVersion}`);
  if (!SERVICE_DELIVERY_BANDS.includes(state.national.overall)) errors.push('invalid national band');
  if (state.regions.length > V1014_CORE_REGIONS.length) errors.push('too many regions');
  return { valid: errors.length === 0, errors };
}

export function runV1014PublicServicesSelfTests(): {
  passed: number;
  failed: number;
  total: number;
  results: Array<{ name: string; passed: boolean; detail: string }>;
} {
  const results: Array<{ name: string; passed: boolean; detail: string }> = [];
  const check = (name: string, ok: boolean, detail: string) => results.push({ name, passed: ok, detail });

  const baseInputs = (): PublicServicesSystemInputs => ({
    turn: 3,
    enabled: true,
    stateCapacity: { national: { deliveryReadiness: 66, coordinationLoad: 48 } },
    publicFinance: { national: { fiscalHeadroomBand: 'balanced' } },
    households: { national: { essentialAccess: 'adequate' } },
    publicStabilityScore: 72,
    completedInfrastructureSupport: 70,
    activeCrises: 0
  });

  check('PS01 schema version', V1014_SCHEMA_VERSION === '10.14.0', V1014_SCHEMA_VERSION);
  check('PS02 five domains', PUBLIC_SERVICE_DOMAINS.length === 5, String(PUBLIC_SERVICE_DOMAINS.length));
  check('PS03 disabled off state', computeNationalPublicServices({ turn: 1, enabled: false }).state.enabled === false, 'enabled=false');

  const r1 = computeNationalPublicServices(baseInputs(), createInitialPublicServicesPersisted());
  check('PS04 compute returns state', Boolean(r1.state.national.summary), r1.state.national.summary);
  check('PS05 validate clean', validatePublicServicesState(r1.state).valid, validatePublicServicesState(r1.state).errors.join(';'));

  const r2 = computeNationalPublicServices(baseInputs(), r1.persisted, r1.state);
  check('PS06 revision increments', r2.persisted.revision === r1.persisted.revision + 1, `${r1.persisted.revision}->${r2.persisted.revision}`);

  const crisisInp = { ...baseInputs(), activeCrises: 3 };
  const rc = computeNationalPublicServices(crisisInp, createInitialPublicServicesPersisted());
  const rn = computeNationalPublicServices(baseInputs(), createInitialPublicServicesPersisted());
  check(
    'PS07 crises reduce access',
    rc.state.national.accessScore <= rn.state.national.accessScore,
    `${rc.state.national.accessScore} vs ${rn.state.national.accessScore}`
  );

  const lowRead = { ...baseInputs(), stateCapacity: { national: { deliveryReadiness: 30, coordinationLoad: 80 } } };
  const rh = computeNationalPublicServices(baseInputs(), createInitialPublicServicesPersisted());
  const rl = computeNationalPublicServices(lowRead, createInitialPublicServicesPersisted());
  check(
    'PS08 low readiness hurts delivery',
    rl.state.national.accessScore < rh.state.national.accessScore,
    `${rl.state.national.accessScore} vs ${rh.state.national.accessScore}`
  );

  const dA = computeNationalPublicServices(baseInputs(), createInitialPublicServicesPersisted());
  const dB = computeNationalPublicServices(baseInputs(), createInitialPublicServicesPersisted());
  check('PS09 determinism', JSON.stringify(dA.state.national) === JSON.stringify(dB.state.national), 'national snapshot match');

  const san = sanitizePublicServicesPersisted({ schemaVersion: '10.14.0', revision: 2, domainBands: r1.persisted.domainBands });
  check('PS10 sanitize roundtrip', san?.revision === 2, String(san?.revision));

  check('PS11 eight regions', r1.state.regions.length === 8, String(r1.state.regions.length));
  check('PS12 hash stable', publicServicesInputHash(baseInputs()) === publicServicesInputHash(baseInputs()), 'hash');

  const built = buildPublicServicesInputs({ turn: 2, gameSettings: { publicServicesEnabled: true, nationalSystemsEnabled: true } });
  check('PS13 build enabled', built.enabled === true, String(built.enabled));

  const passed = results.filter(r => r.passed).length;
  return { passed, failed: results.length - passed, total: results.length, results };
}

export function sanitizePolicySystemsPersisted(raw: unknown, turn = 1): PolicySystemsPersisted | null {
  if (!raw || typeof raw !== 'object') return null;
  const p = raw as Partial<PolicySystemsPersisted>;
  const base = createInitialPolicySystemsPersisted(turn);
  const themes: NationalPolicyAgendaTheme[] = ['growth', 'stability', 'equity', 'climate', 'infrastructure', 'innovation'];
  const pickTheme = (v: unknown, fallback: NationalPolicyAgendaTheme) =>
    themes.includes(v as NationalPolicyAgendaTheme) ? (v as NationalPolicyAgendaTheme) : fallback;
  const agendaRaw = (p.agenda || {}) as Partial<NationalPolicyAgendaState>;
  const pinned = Array.isArray(agendaRaw.pinnedThemes)
    ? agendaRaw.pinnedThemes.filter(t => themes.includes(t)).slice(0, 3)
    : [];
  return {
    ...base,
    schemaVersion: typeof p.schemaVersion === 'string' ? p.schemaVersion : V1014_SCHEMA_VERSION,
    revision: typeof p.revision === 'number' ? Math.max(0, Math.floor(p.revision)) : 0,
    agenda: {
      primary: pickTheme(agendaRaw.primary, base.agenda.primary),
      secondary: agendaRaw.secondary ? pickTheme(agendaRaw.secondary, base.agenda.secondary!) : null,
      pinnedThemes: pinned
    },
    lastDomainStances: typeof p.lastDomainStances === 'object' && p.lastDomainStances ? p.lastDomainStances : {},
    lastCoherence: typeof p.lastCoherence === 'number' ? p.lastCoherence : 50,
    lastInputHash: typeof p.lastInputHash === 'string' ? p.lastInputHash : '',
    lastUpdatedTurn: typeof p.lastUpdatedTurn === 'number' ? p.lastUpdatedTurn : turn
  };
}


function v1014StanceFromScore(score: number): PolicyStanceBand {
  if (score < 22) return 'restrictive';
  if (score < 38) return 'cautious';
  if (score < 58) return 'balanced';
  if (score < 78) return 'supportive';
  return 'expansionary';
}

function v1014FiscalScore(p: FiscalPressureBand | null, headroom: NationalFiscalPosition['headroomBand'] | null): number {
  const pressureMap: Record<FiscalPressureBand, number> = {
    comfortable: 82,
    healthy: 68,
    tight: 48,
    strained: 32,
    critical: 18
  };
  let s = p ? pressureMap[p] : 50;
  if (headroom === 'high') s += 8;
  if (headroom === 'critical') s -= 12;
  else if (headroom === 'low') s -= 6;
  return v1014Clamp(s, 0, 100);
}

function v1014RegulatoryScore(pressure: StateCapacityPressureBand | null): number {
  if (!pressure) return 50;
  const map: Record<StateCapacityPressureBand, number> = {
    underused: 72,
    healthy: 62,
    busy: 52,
    strained: 36,
    overloaded: 22,
    critical: 12
  };
  return map[pressure] ?? 50;
}

function v1014SocialScore(prosperity: LivingStandardsBand | null, stability: number): number {
  const bandMap: Record<LivingStandardsBand, number> = {
    stressed: 22,
    weak: 35,
    moderate: 52,
    strong: 72,
    prosperous: 86
  };
  let s = prosperity ? bandMap[prosperity] : 50;
  s += v1014Clamp((stability - 50) * 0.35, -12, 12);
  return v1014Clamp(s, 0, 100);
}

function v1014EnvironmentalScore(priorities: string[], riskPressure: number): number {
  const climateFocus = priorities.some(p => /climate|environment|water|energy transition/i.test(p));
  let s = climateFocus ? 68 : 48;
  s -= v1014Clamp(riskPressure * 0.25, 0, 18);
  return v1014Clamp(s, 0, 100);
}

function v1014TradeScore(priorities: string[], fiscalScore: number): number {
  const tradeFocus = priorities.some(p => /trade|export|logistics|port/i.test(p));
  let s = tradeFocus ? 70 : 52;
  s += (fiscalScore - 50) * 0.12;
  return v1014Clamp(s, 0, 100);
}

function v1014SecurityScore(stability: number, riskPressure: number): number {
  let s = 40 + stability * 0.45;
  s -= v1014Clamp(riskPressure * 0.35, 0, 22);
  return v1014Clamp(s, 0, 100);
}

const V1014_THEME_DOMAIN_WEIGHTS: Record<NationalPolicyAgendaTheme, Partial<Record<NationalPolicyDomain, number>>> = {
  growth: { fiscal: 0.35, trade: 0.3, regulatory: 0.15, social: 0.1, environmental: 0.05, security: 0.05 },
  stability: { security: 0.3, fiscal: 0.25, regulatory: 0.2, social: 0.15, environmental: 0.05, trade: 0.05 },
  equity: { social: 0.4, fiscal: 0.2, regulatory: 0.15, environmental: 0.1, trade: 0.1, security: 0.05 },
  climate: { environmental: 0.45, regulatory: 0.2, fiscal: 0.15, social: 0.1, trade: 0.05, security: 0.05 },
  infrastructure: { fiscal: 0.3, trade: 0.25, regulatory: 0.2, environmental: 0.1, social: 0.1, security: 0.05 },
  innovation: { regulatory: 0.25, fiscal: 0.25, trade: 0.2, social: 0.15, environmental: 0.1, security: 0.05 }
};

function v1014AgendaTargetStance(theme: NationalPolicyAgendaTheme): number {
  const targets: Record<NationalPolicyAgendaTheme, number> = {
    growth: 72,
    stability: 55,
    equity: 62,
    climate: 58,
    infrastructure: 68,
    innovation: 70
  };
  return targets[theme];
}

function v1014DomainAgendaCoherence(domain: NationalPolicyDomain, score: number, agenda: NationalPolicyAgendaState): number {
  const themes = [agenda.primary, agenda.secondary, ...agenda.pinnedThemes].filter(Boolean) as NationalPolicyAgendaTheme[];
  if (!themes.length) return 50;
  let weightedTarget = 0;
  let wSum = 0;
  for (const t of themes) {
    const w = V1014_THEME_DOMAIN_WEIGHTS[t][domain] ?? 0.15;
    weightedTarget += v1014AgendaTargetStance(t) * w;
    wSum += w;
  }
  const target = wSum > 0 ? weightedTarget / wSum : 55;
  return v1014Clamp(100 - Math.abs(score - target), 0, 100);
}

export function policySystemsInputHash(inputs: PolicySystemsInputs, persisted: PolicySystemsPersisted | null): string {
  const raw = JSON.stringify([
    inputs.turn,
    inputs.enabled,
    Math.round(inputs.publicStability),
    inputs.nationalPrioritiesTop.slice(0, 5),
    Math.round(inputs.systemicRiskPressure),
    inputs.fiscalPressure,
    inputs.fiscalHeadroomBand,
    inputs.stateCapacityPressure,
    inputs.householdProsperityBand,
    persisted?.agenda || inputs.agendaOverride || null
  ]);
  let h = 2166136261;
  for (let k = 0; k < raw.length; k++) { h ^= raw.charCodeAt(k); h = Math.imul(h, 16777619); }
  return (h >>> 0).toString(36);
}

export function buildPolicySystemsInputs(params: {
  turn: number;
  gameSettings: any;
  gameState: any;
  publicFinance?: NationalPublicFinanceState | null;
  stateCapacity?: NationalStateCapacityState | null;
  households?: NationalHouseholdEconomyState | null;
  nationalPriorities?: NationalPrioritiesState | null;
  systemicRisk?: SystemicRiskState | null;
  persisted?: PolicySystemsPersisted | null;
}): PolicySystemsInputs {
  const gset = params.gameSettings || {};
  const enabled = gset.nationalSystemsEnabled !== false && gset.policySystemsEnabled !== false;
  const pf = params.publicFinance || null;
  const sc = params.stateCapacity || null;
  const hh = params.households || null;
  const np = params.nationalPriorities || null;
  const sr = params.systemicRisk || null;
  const stability = Number(params.gameState?.publicStabilityState?.nationalScore ?? params.gameState?.publicStability ?? 70);
  const prioritiesTop = (np?.activePriorities || []).slice(0, 5).map((p: any) => String(p.title || p.id || ''));
  return {
    turn: params.turn,
    enabled,
    publicStability: stability,
    nationalPrioritiesTop: prioritiesTop,
    systemicRiskPressure: Number(sr?.nationalPressure?.pressureScore ?? sr?.pressureScore ?? 20),
    fiscalPressure: pf?.fiscalPosition?.pressure ?? null,
    fiscalHeadroomBand: pf?.fiscalPosition?.headroomBand ?? null,
    stateCapacityPressure: sc?.implementationLoad?.pressure ?? null,
    householdProsperityBand: hh?.national?.prosperity ?? null,
    agendaOverride: params.persisted?.agenda
  };
}

export function computeNationalPolicySystems(
  inputs: PolicySystemsInputs,
  prev: PolicySystemsPersisted | null
): PolicySystemsComputeResult {
  const persistedBase = prev ? { ...prev } : createInitialPolicySystemsPersisted(inputs.turn);
  const agenda: NationalPolicyAgendaState = {
    ...persistedBase.agenda,
    ...(inputs.agendaOverride || {})
  };
  const inputHash = policySystemsInputHash(inputs, persistedBase);
  if (!inputs.enabled) {
    const emptyDomains = {} as Record<NationalPolicyDomain, PolicyDomainState>;
    for (const d of NATIONAL_POLICY_DOMAINS) {
      emptyDomains[d] = {
        domain: d,
        stance: 'balanced',
        score: 50,
        coherenceWithAgenda: 50,
        evidence: [],
        summary: 'Policy systems disabled.'
      };
    }
    return {
      state: {
        schemaVersion: V1014_SCHEMA_VERSION,
        revision: persistedBase.revision,
        turn: inputs.turn,
        enabled: false,
        agenda,
        domains: emptyDomains,
        coherence: { overall: 50, agendaAlignment: 50, crossDomainFriction: 0, bindingConstraint: null, explanation: 'Disabled.' },
        inputHash,
        lastUpdatedTurn: inputs.turn
      },
      persisted: { ...persistedBase, lastInputHash: inputHash, lastUpdatedTurn: inputs.turn }
    };
  }

  const fiscalScore = v1014FiscalScore(inputs.fiscalPressure, inputs.fiscalHeadroomBand);
  const domainScores: Record<NationalPolicyDomain, number> = {
    fiscal: fiscalScore,
    regulatory: v1014RegulatoryScore(inputs.stateCapacityPressure),
    social: v1014SocialScore(inputs.householdProsperityBand, inputs.publicStability),
    environmental: v1014EnvironmentalScore(inputs.nationalPrioritiesTop, inputs.systemicRiskPressure),
    trade: v1014TradeScore(inputs.nationalPrioritiesTop, fiscalScore),
    security: v1014SecurityScore(inputs.publicStability, inputs.systemicRiskPressure)
  };

  const domains = {} as Record<NationalPolicyDomain, PolicyDomainState>;
  let frictionSum = 0;
  for (const d of NATIONAL_POLICY_DOMAINS) {
    const score = domainScores[d];
    const stance = v1014StanceFromScore(score);
    const coherenceWithAgenda = v1014DomainAgendaCoherence(d, score, agenda);
    if (persistedBase.lastDomainStances[d] && persistedBase.lastDomainStances[d] !== stance) {
      frictionSum += Math.abs(POLICY_STANCE_RANK[stance] - POLICY_STANCE_RANK[persistedBase.lastDomainStances[d]!]);
    }
    domains[d] = {
      domain: d,
      stance,
      score,
      coherenceWithAgenda,
      evidence: [
        { id: `${d}_core`, sourceSystem: 'v1014', factor: 'derived_posture', contribution: score, turn: inputs.turn }
      ],
      summary: `${POLICY_STANCE_LABEL[stance]} posture (${Math.round(score)}/100).`
    };
  }

  const agendaAlignment =
    NATIONAL_POLICY_DOMAINS.reduce((s, d) => s + domains[d].coherenceWithAgenda, 0) / NATIONAL_POLICY_DOMAINS.length;
  const crossDomainFriction = v1014Clamp(frictionSum * 4, 0, 40);
  const overall = v1014Clamp(agendaAlignment - crossDomainFriction * 0.35, 0, 100);

  let bindingConstraint: NationalPolicyDomain | null = null;
  let lowestHeadroom = 999;
  if (inputs.fiscalPressure === 'critical' || inputs.fiscalHeadroomBand === 'critical') {
    bindingConstraint = 'fiscal';
  } else if (inputs.stateCapacityPressure === 'overloaded' || inputs.stateCapacityPressure === 'strained') {
    bindingConstraint = 'regulatory';
  } else {
    for (const d of NATIONAL_POLICY_DOMAINS) {
      const gap = 100 - domains[d].coherenceWithAgenda;
      if (gap > lowestHeadroom) {
        lowestHeadroom = gap;
        bindingConstraint = d;
      }
    }
  }

  const lastStances: Partial<Record<NationalPolicyDomain, PolicyStanceBand>> = {};
  for (const d of NATIONAL_POLICY_DOMAINS) lastStances[d] = domains[d].stance;

  const persisted: PolicySystemsPersisted = {
    ...persistedBase,
    revision: persistedBase.revision + (persistedBase.lastInputHash === inputHash ? 0 : 1),
    agenda,
    lastDomainStances: lastStances,
    lastCoherence: overall,
    lastInputHash: inputHash,
    lastUpdatedTurn: inputs.turn
  };

  return {
    state: {
      schemaVersion: V1014_SCHEMA_VERSION,
      revision: persisted.revision,
      turn: inputs.turn,
      enabled: true,
      agenda,
      domains,
      coherence: {
        overall,
        agendaAlignment,
        crossDomainFriction,
        bindingConstraint,
        explanation:
          bindingConstraint === 'fiscal'
            ? 'Fiscal headroom is the binding constraint on the national agenda.'
            : bindingConstraint === 'regulatory'
              ? 'Institutional delivery capacity is limiting policy execution.'
              : `Largest agenda gap sits in ${bindingConstraint || 'cross-domain'} alignment.`
      },
      inputHash,
      lastUpdatedTurn: inputs.turn
    },
    persisted
  };
}

export function validateNationalPolicySystemsState(state: NationalPolicySystemsState): { valid: boolean; issues: string[] } {
  const issues: string[] = [];
  if (state.schemaVersion !== V1014_SCHEMA_VERSION) issues.push('schema mismatch');
  for (const d of NATIONAL_POLICY_DOMAINS) {
    if (!state.domains[d]) issues.push(`missing domain ${d}`);
  }
  if (!Number.isFinite(state.coherence.overall)) issues.push('coherence non-finite');
  return { valid: issues.length === 0, issues };
}

export type PolicySystemsQueryTopic = 'overview' | 'agenda' | 'domain' | 'coherence';

  topic: PolicySystemsQueryTopic;
  domain: NationalPolicyDomain | null;
}

export function detectPolicySystemsQuery(raw: string, gw: GIWorld): PolicySystemsQuery | null {
  const view = (gw.national as any)?.policySystems as PolicySystemsWorldView | undefined;
  if (!view?.enabled || !view.state?.enabled) return null;
  const q = ` ${String(raw || '').toLowerCase().replace(/[’']/g, "'").replace(/[?!.]/g, ' ').replace(/\s+/g, ' ').trim()} `;
  if (!/\b(policy|policies|government agenda|national agenda|policy stance|policy posture)\b/.test(q)) return null;
  const domain =
    (/\bfiscal\b/.test(q) && 'fiscal') ||
    (/\b(regulation|regulatory|red tape)\b/.test(q) && 'regulatory') ||
    (/\b(social|equity|services|household)\b/.test(q) && 'social') ||
    (/\b(climate|environment|environmental)\b/.test(q) && 'environmental') ||
    (/\b(trade|export|openness)\b/.test(q) && 'trade') ||
    (/\b(security|defence|defense)\b/.test(q) && 'security') ||
    null;
  if (/\b(agenda|priority|priorities|platform)\b/.test(q)) return { topic: 'agenda', domain: null };
  if (/\b(coherence|alignment|friction|constraint)\b/.test(q)) return { topic: 'coherence', domain: null };
  if (domain) return { topic: 'domain', domain };
  return { topic: 'overview', domain: null };
}

export function composePolicySystemsAnswer(query: PolicySystemsQuery, gw: GIWorld): GIComposePart & { shape: GIAnswerShape } {
  const view = (gw.national as any)!.policySystems as PolicySystemsWorldView;
  const s = view.state;
  const sections: GIAnswerSection[] = [];
  const say = (id: string, heading: string | null, text: string) => {
    sections.push({ id, heading, claims: [claim(text, 'calculated', 'high', ['policy.systems'])] });
  };
  let title = 'National policy systems';
  switch (query.topic) {
    case 'agenda':
      title = 'Government policy agenda';
      say(
        'agenda',
        'Agenda',
        `Primary: ${POLICY_AGENDA_THEME_LABEL[s.agenda.primary]}${
          s.agenda.secondary ? `; secondary: ${POLICY_AGENDA_THEME_LABEL[s.agenda.secondary]}` : ''
        }. Coherence ${Math.round(s.coherence.overall)}/100.`
      );
      break;
    case 'coherence':
      title = 'Policy coherence';
      say(
        'coherence',
        'Coherence',
        `${s.coherence.explanation} Overall ${Math.round(s.coherence.overall)}/100 (agenda ${Math.round(
          s.coherence.agendaAlignment
        )}, friction ${Math.round(s.coherence.crossDomainFriction)}).`
      );
      break;
    case 'domain': {
      const d = query.domain || 'fiscal';
      const dom = s.domains[d];
      title = `${d} policy stance`;
      say('domain', 'Domain', `${dom.summary} Agenda fit ${Math.round(dom.coherenceWithAgenda)}/100.`);
      break;
    }
    default:
      say(
        'overview',
        'Overview',
        `National posture is ${POLICY_STANCE_LABEL[s.domains.fiscal.stance]} fiscally and ${POLICY_STANCE_LABEL[s.domains.social.stance]} on social policy; coherence ${Math.round(
          s.coherence.overall
        )}/100.`
      );
  }
  return { title, sections, shape: 'status' };
}

export function policySystemsActionAlignment(
  state: NationalPolicySystemsState | null,
  action: { kind?: string; category?: string; theme?: string }
): { score: number; reason: string | null } {
  if (!state?.enabled) return { score: 0, reason: null };
  const kind = String(action.kind || action.category || '').toLowerCase();
  let domain: NationalPolicyDomain = 'fiscal';
  if (/environment|climate|water/.test(kind)) domain = 'environmental';
  else if (/trade|export|port/.test(kind)) domain = 'trade';
  else if (/security|defense|defence/.test(kind)) domain = 'security';
  else if (/social|housing|health/.test(kind)) domain = 'social';
  else if (/regul|permit|compliance/.test(kind)) domain = 'regulatory';
  const dom = state.domains[domain];
  const agendaBoost = dom.coherenceWithAgenda >= 70 ? 1.5 : dom.coherenceWithAgenda <= 35 ? -1.5 : 0;
  const stanceBoost = (POLICY_STANCE_RANK[dom.stance] - 2) * 0.35;
  const score = v1014Clamp(agendaBoost + stanceBoost, -2, 2);
  return { score, reason: score !== 0 ? `${domain} policy alignment ${score > 0 ? '+' : ''}${score.toFixed(1)}` : null };
}

// --- 20AG-wellbeing: NATIONAL WELLBEING INDEX ---
// ============================================================================
// Derived interpretation layer only — composes V10.13 living standards with optional
// Public Stability scores. Never mutates households, stability, Living Regions, or fiscal state.
// Invariants W1–W6:
//   W1 — V10.13 remains authoritative for material living standards and domestic demand.
//   W2 — Public Stability remains authoritative for societal stability; V10.14 reads scores only.
//   W3 — No wellbeing mana, no household actors, no direct economy or AP effects.
//   W4 — Feature off when nationalSystemsEnabled, householdsEnabled, or nationalWellbeingEnabled is false.
//   W5 — Bands use hysteresis on the composite score (no raw-formula flicker).
//   W6 — Pre-V10.14 saves load with null persisted memory; derived bands only.


export type NationalWellbeingBand = 'critical' | 'strained' | 'mixed' | 'positive' | 'flourishing';

export const NATIONAL_WELLBEING_LABEL: Record<NationalWellbeingBand, string> = {
  critical: 'Critical wellbeing',
  strained: 'Strained wellbeing',
  mixed: 'Mixed wellbeing',
  positive: 'Positive wellbeing',
  flourishing: 'Flourishing wellbeing'
};

export const NATIONAL_WELLBEING_ICON: Record<NationalWellbeingBand, string> = {
  critical: '⛔',
  strained: '●',
  mixed: '◐',
  positive: '✓',
  flourishing: '★'
};

const V1014_BAND_BOUNDS: Array<[NationalWellbeingBand, number]> = [
  ['critical', 0],
  ['strained', 28],
  ['mixed', 45],
  ['positive', 58],
  ['flourishing', 75]
];

const v1014C = (n: number) => Math.round(v1014Clamp(n) * 10) / 10;

function v1014BandForScore(score: number): NationalWellbeingBand {
  let band: NationalWellbeingBand = 'critical';
  for (const [b, min] of V1014_BAND_BOUNDS) {
    if (score >= min) band = b;
  }
  return band;
}

function v1014BandWithHysteresis(score: number, prev: { score: number; band: NationalWellbeingBand } | null): NationalWellbeingBand {
  const raw = v1014BandForScore(score);
  if (!prev) return raw;
  const dead = V1013_LIMITS.hysteresisDeadband;
  const prevIdx = V1014_BAND_BOUNDS.findIndex(([b]) => b === prev.band);
  const rawIdx = V1014_BAND_BOUNDS.findIndex(([b]) => b === raw);
  if (rawIdx === prevIdx) return prev.band;
  if (rawIdx > prevIdx && score < V1014_BAND_BOUNDS[rawIdx][1] + dead) return prev.band;
  if (rawIdx < prevIdx && score > V1014_BAND_BOUNDS[prevIdx][1] - dead) return prev.band;
  return raw;
}

function v1014NormRank(rank: number, max: number): number {
  return max <= 0 ? 50 : v1014C((rank / max) * 100);
}

function v1014InverseRank(rank: number, max: number): number {
  return v1014NormRank(max - rank, max);
}

export function sanitizeNationalWellbeingPersisted(raw: unknown, turn = 0): NationalWellbeingPersisted {
  const base = createInitialNationalWellbeingPersisted();
  if (!raw || typeof raw !== 'object') return base;
  const o = raw as Record<string, unknown>;
  const band = (o.bandHysteresis && typeof o.bandHysteresis === 'object' ? o.bandHysteresis : null) as Record<string, unknown> | null;
  const bandName = band && typeof band.band === 'string' && band.band in NATIONAL_WELLBEING_LABEL ? (band.band as NationalWellbeingBand) : base.bandHysteresis.band;
  const score = band && typeof band.score === 'number' ? v1014C(band.score) : base.bandHysteresis.score;
  return {
    schemaVersion: typeof o.schemaVersion === 'string' ? o.schemaVersion : V1014_SCHEMA_VERSION,
    revision: typeof o.revision === 'number' ? o.revision : 0,
    initializedTurn: typeof o.initializedTurn === 'number' ? o.initializedTurn : null,
    lastUpdatedTurn: typeof o.lastUpdatedTurn === 'number' ? o.lastUpdatedTurn : turn,
    lastInputHash: typeof o.lastInputHash === 'string' ? o.lastInputHash : '',
    bandHysteresis: { score, band: bandName }
  };
}

export function nationalWellbeingInputHash(inputs: NationalWellbeingInputs): string {
  const hh = inputs.households;
  const snap = hh
    ? `${hh.inputHash}|${hh.national.prosperity}|${hh.national.affordability}|${hh.national.trajectory}|${Object.keys(hh.regions).length}`
    : 'none';
  const stab = inputs.publicStabilityScore == null ? 'na' : String(Math.round(inputs.publicStabilityScore));
  return `v1014:${inputs.turn}:${inputs.enabled ? 1 : 0}:${snap}:${stab}`;
}

export function buildNationalWellbeingInputs(args: {
  turn: number;
  enabled: boolean;
  householdState: NationalHouseholdEconomyState | null;
  publicStabilityState?: PublicStabilityState | null;
  publicStabilityEnabled?: boolean;
}): NationalWellbeingInputs {
  const stabilityOn = args.publicStabilityEnabled !== false;
  const score =
    stabilityOn && args.publicStabilityState && typeof args.publicStabilityState.nationalStability === 'number'
      ? v1014C(args.publicStabilityState.nationalStability)
      : null;
  return {
    turn: args.turn,
    enabled: args.enabled,
    households: args.householdState,
    publicStabilityScore: score
  };
}

function v1014RegionalProsperityScore(reg: RegionalLivingStandardsState): number {
  const pr = V1013_PROSPERITY_RANK[reg.prosperity] ?? 2;
  const af = V1013_AFFORDABILITY_RANK[reg.affordability] ?? 2;
  const hp = V1013_HOUSING_PRESSURE_RANK[reg.housingPressure] ?? 2;
  return v1014C(pr * 8 + af * 6 + v1014InverseRank(hp, 4) * 4);
}

export function computeNationalWellbeing(
  inputs: NationalWellbeingInputs,
  persistedIn: NationalWellbeingPersisted | null,
  _prevState: NationalWellbeingState | null
): NationalWellbeingComputeResult {
  let persisted = persistedIn ? sanitizeNationalWellbeingPersisted(persistedIn, inputs.turn) : createInitialNationalWellbeingPersisted();
  const inputHash = nationalWellbeingInputHash(inputs);
  if (!inputs.enabled || !inputs.households) {
    return { state: null, persisted: { ...persisted, lastInputHash: inputHash, lastUpdatedTurn: inputs.turn } };
  }
  if (persisted.lastInputHash === inputHash && persisted.revision > 0) {
    const band = persisted.bandHysteresis.band;
    const score = persisted.bandHysteresis.score;
    return {
      state: {
        schemaVersion: V1014_SCHEMA_VERSION,
        revision: persisted.revision,
        enabled: true,
        score,
        band,
        dimensions: {
          material: score,
          security: score,
          opportunity: score,
          housing: score,
          cohesion: inputs.publicStabilityScore
        },
        trajectory: inputs.households.national.trajectory,
        worstRegionId: null,
        bestRegionId: null,
        divergenceBand: 'low',
        concerns: [],
        strengths: [],
        summary: NATIONAL_WELLBEING_LABEL[band],
        inputHash,
        lastUpdatedTurn: inputs.turn,
        sources: { households: true, publicStability: inputs.publicStabilityScore != null }
      },
      persisted
    };
  }

  const nat = inputs.households.national;
  const material = v1014C(
    (v1014NormRank(V1013_PROSPERITY_RANK[nat.prosperity], 4) * 0.34 +
      v1014NormRank(V1013_AFFORDABILITY_RANK[nat.affordability], 4) * 0.33 +
      v1014NormRank(V1013_ESSENTIAL_ACCESS_RANK[nat.essentialAccess], 4) * 0.33)
  );
  const secRank = Math.max(0, V1013_SECURITY_ORDER.indexOf(nat.economicSecurity));
  const security = v1014C(v1014NormRank(secRank, V1013_SECURITY_ORDER.length - 1) * 0.5 + v1014NormRank(V1013_RESILIENCE_RANK[nat.financialResilience], 4) * 0.5);
  const opportunity = v1014NormRank(V1013_EMPLOYMENT_RANK[nat.employmentOpportunity], 4);
  const housing = v1014InverseRank(V1013_HOUSING_PRESSURE_RANK[nat.housingPressure], 4);
  const cohesion = inputs.publicStabilityScore == null ? null : v1014C(inputs.publicStabilityScore);

  let composite = material * 0.28 + security * 0.2 + opportunity * 0.18 + housing * 0.14;
  if (cohesion != null) composite += cohesion * 0.2;
  else composite /= 0.8;
  const score = v1014C(composite);
  const band = v1014BandWithHysteresis(score, persisted.bandHysteresis);

  const regionalScores = Object.entries(inputs.households.regions).map(([id, reg]) => ({ id, s: v1014RegionalProsperityScore(reg) }));
  regionalScores.sort((a, b) => a.s - b.s);
  const worstRegionId = regionalScores[0]?.id || null;
  const bestRegionId = regionalScores[regionalScores.length - 1]?.id || null;
  const spread = regionalScores.length ? regionalScores[regionalScores.length - 1].s - regionalScores[0].s : 0;
  const divergenceBand: NationalWellbeingState['divergenceBand'] = spread >= 28 ? 'high' : spread >= 14 ? 'moderate' : 'low';

  const concerns: string[] = [];
  if (nat.prosperity === 'stressed' || nat.prosperity === 'weak') concerns.push('Material living standards are under pressure nationally.');
  if (nat.affordability === 'severe_pressure' || nat.affordability === 'strained') concerns.push('Affordability strain is weighing on households.');
  if (nat.housingPressure === 'severe' || nat.housingPressure === 'high') concerns.push('Housing pressure is elevated in key regions.');
  if (cohesion != null && cohesion < 45) concerns.push('Public stability is weak, dragging social cohesion.');
  if (divergenceBand === 'high' && worstRegionId) concerns.push(`Wellbeing diverges sharply — ${worstRegionId} lags the national average.`);

  const strengths: string[] = [];
  if (nat.prosperity === 'strong' || nat.prosperity === 'prosperous') strengths.push('Broad-based prosperity supports wellbeing.');
  if (nat.employmentOpportunity === 'strong' || nat.employmentOpportunity === 'abundant') strengths.push('Employment opportunity is supportive.');
  if (cohesion != null && cohesion >= 75) strengths.push('Social cohesion from stability is a buffer.');
  if (nat.trajectory === 'improving') strengths.push('Living standards trajectory is improving.');

  const summary = `${NATIONAL_WELLBEING_LABEL[band]} (${Math.round(score)}/100) — ${concerns[0] || strengths[0] || 'Conditions are broadly balanced.'}`;

  persisted = {
    ...persisted,
    revision: persisted.revision + 1,
    initializedTurn: persisted.initializedTurn ?? inputs.turn,
    lastUpdatedTurn: inputs.turn,
    lastInputHash: inputHash,
    bandHysteresis: { score, band }
  };

  const state: NationalWellbeingState = {
    schemaVersion: V1014_SCHEMA_VERSION,
    revision: persisted.revision,
    enabled: true,
    score,
    band,
    dimensions: { material, security, opportunity, housing, cohesion },
    trajectory: nat.trajectory,
    worstRegionId,
    bestRegionId,
    divergenceBand,
    concerns: concerns.slice(0, 4),
    strengths: strengths.slice(0, 4),
    summary,
    inputHash,
    lastUpdatedTurn: inputs.turn,
    sources: { households: true, publicStability: cohesion != null }
  };

  return { state, persisted };
}

export function validateNationalWellbeingState(state: NationalWellbeingState | null): { valid: boolean; issues: string[] } {
  if (!state) return { valid: true, issues: [] };
  const issues: string[] = [];
  if (state.schemaVersion !== V1014_SCHEMA_VERSION) issues.push('schemaVersion mismatch');
  if (state.score < 0 || state.score > 100) issues.push('score out of range');
  if (!(state.band in NATIONAL_WELLBEING_LABEL)) issues.push('invalid band');
  return { valid: issues.length === 0, issues };
}

      makeDefaultV1014TestHouseholdInputs({ highInflationScenarioActive: true }, { NSW: { economyBand: 'weak', housingNeedSeverity: 'critical', investmentPressure: 'overheated' } }),
      createInitialHouseholdEconomyPersisted(),
      null
    ).state;
    const r = computeNationalWellbeing({ turn: 4, enabled: true, households: hh, publicStabilityScore: 70 }, createInitialNationalWellbeingPersisted(), null);
    return r.state != null && (r.state.band === 'strained' || r.state.band === 'critical' || r.state.band === 'mixed');
  })(), `band=${(() => {
    const hh = computeNationalHouseholdEconomy(
      makeDefaultV1014TestHouseholdInputs({ highInflationScenarioActive: true }, { NSW: { economyBand: 'weak', housingNeedSeverity: 'critical' } }),
      createInitialHouseholdEconomyPersisted(),
      null
    ).state;
    return computeNationalWellbeing({ turn: 4, enabled: true, households: hh, publicStabilityScore: 70 }, createInitialNationalWellbeingPersisted(), null).state?.band;
  })()}`);

  check('TEST 4 — COHESION READ ONLY', (() => {
    const hh = computeNationalHouseholdEconomy(makeDefaultV1014TestHouseholdInputs(), createInitialHouseholdEconomyPersisted(), null).state;
    const before = hh.national.prosperity;
    computeNationalWellbeing({ turn: 2, enabled: true, households: hh, publicStabilityScore: 22 }, createInitialNationalWellbeingPersisted(), null);
    return hh.national.prosperity === before;
  })(), 'household state unchanged');

  check('TEST 5 — NO STABILITY DATA', (() => {
    const hh = computeNationalHouseholdEconomy(makeDefaultV1014TestHouseholdInputs(), createInitialHouseholdEconomyPersisted(), null).state;
    const r = computeNationalWellbeing({ turn: 2, enabled: true, households: hh, publicStabilityScore: null }, createInitialNationalWellbeingPersisted(), null);
    return r.state != null && r.state.dimensions.cohesion === null && r.state.sources.publicStability === false;
  })(), 'cohesion null without stability');

  check('TEST 6 — LOW STABILITY DRAG', (() => {
    const hh = computeNationalHouseholdEconomy(makeDefaultV1014TestHouseholdInputs(), createInitialHouseholdEconomyPersisted(), null).state;
    const high = computeNationalWellbeing({ turn: 2, enabled: true, households: hh, publicStabilityScore: 85 }, createInitialNationalWellbeingPersisted(), null).state!;
    const low = computeNationalWellbeing({ turn: 2, enabled: true, households: hh, publicStabilityScore: 25 }, createInitialNationalWellbeingPersisted(), null).state!;
    return high.score > low.score;
  })(), 'lower stability lowers composite');

  check('TEST 7 — REGIONAL DIVERGENCE', (() => {
    const hh = computeNationalHouseholdEconomy(
      makeDefaultV1014TestHouseholdInputs({}, { NSW: { economyBand: 'strong', essentialInfraScore: 90 }, WA: { economyBand: 'weak', essentialInfraScore: 35, housingNeedSeverity: 'critical' } }),
      createInitialHouseholdEconomyPersisted(),

  check('TEST 9 — SAVE LOAD', (() => {
    const hh = computeNationalHouseholdEconomy(makeDefaultV1014TestHouseholdInputs({ turn: 7 }), createInitialHouseholdEconomyPersisted(), null).state;
    const run1 = computeNationalWellbeing({ turn: 7, enabled: true, households: hh, publicStabilityScore: 66 }, createInitialNationalWellbeingPersisted(), null);
    const restored = sanitizeNationalWellbeingPersisted(JSON.parse(JSON.stringify(run1.persisted)));
    const run2 = computeNationalWellbeing({ turn: 7, enabled: true, households: hh, publicStabilityScore: 66 }, restored, null);
    return run1.state?.band === run2.state?.band && run1.state?.score === run2.state?.score;
  })(), 'persisted round-trip');

  check('TEST 10 — OLD SAVE', (() => {
    const hh = computeNationalHouseholdEconomy(makeDefaultV1014TestHouseholdInputs({ turn: 8 }), createInitialHouseholdEconomyPersisted(), null).state;
    const r = computeNationalWellbeing({ turn: 8, enabled: true, households: hh, publicStabilityScore: 55 }, sanitizeNationalWellbeingPersisted(undefined), null);
    return r.state != null && r.persisted.initializedTurn === 8;
  })(), 'null persisted derives cleanly');

  check('TEST 11 — VALIDATION', (() => {
    const hh = computeNationalHouseholdEconomy(makeDefaultV1014TestHouseholdInputs(), createInitialHouseholdEconomyPersisted(), null).state;
    const r = computeNationalWellbeing({ turn: 1, enabled: true, households: hh, publicStabilityScore: 50 }, createInitialNationalWellbeingPersisted(), null);
    return validateNationalWellbeingState(r.state).valid;
  })(), 'validateNationalWellbeingState');

  check('TEST 12 — INPUT HASH CACHE', (() => {
    const hh = computeNationalHouseholdEconomy(makeDefaultV1014TestHouseholdInputs({ turn: 9 }), createInitialHouseholdEconomyPersisted(), null).state;
    const first = computeNationalWellbeing({ turn: 9, enabled: true, households: hh, publicStabilityScore: 72 }, createInitialNationalWellbeingPersisted(), null);
    const second = computeNationalWellbeing({ turn: 9, enabled: true, households: hh, publicStabilityScore: 72 }, first.persisted, first.state);

  check('TEST 13 — HYSTERESIS', (() => {
    const hh = computeNationalHouseholdEconomy(
      makeDefaultV1014TestHouseholdInputs({ highInflationScenarioActive: true }, { NSW: { economyBand: 'weak', housingNeedSeverity: 'critical' } }),
      createInitialHouseholdEconomyPersisted(),

    return r.state != null && r.state.score < 58 && r.state.band === 'positive';
  })(), 'band holds on downgrade inside deadband');

  check('TEST 14 — BUILD INPUTS HELPER', (() => {
    const hh = computeNationalHouseholdEconomy(makeDefaultV1014TestHouseholdInputs(), createInitialHouseholdEconomyPersisted(), null).state;
    const inp = buildNationalWellbeingInputs({ turn: 2, enabled: true, householdState: hh, publicStabilityState: createDefaultPublicStabilityState(), publicStabilityEnabled: true });
    return inp.publicStabilityScore === 80;
  })(), 'reads default stability');

  check('TEST 15 — CONCERNS WHEN STRAINED', (() => {
    const hh = computeNationalHouseholdEconomy(
      makeDefaultV1014TestHouseholdInputs({ highInflationScenarioActive: true }, { VIC: { housingNeedSeverity: 'critical', investmentPressure: 'overheated' } }),
      createInitialHouseholdEconomyPersisted(),
      null
    ).state;
    const r = computeNationalWellbeing({ turn: 4, enabled: true, households: hh, publicStabilityScore: 40 }, createInitialNationalWellbeingPersisted(), null);
    return (r.state?.concerns.length || 0) > 0;
  })(), 'concerns populated under strain');

  const passed = results.filter(r => r.passed).length;
  return { passed, failed: results.length - passed, total: results.length, results };
}
